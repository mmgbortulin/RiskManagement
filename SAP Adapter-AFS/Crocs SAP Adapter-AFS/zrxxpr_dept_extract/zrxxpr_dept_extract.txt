FORM fill_deptstyle .
* DESCRIPTION:  Fill Department/Style HDI table

  CLEAR:  gs_deptstyle,
          gt_deptstyle[].
  CLEAR:  gv_seq,
          gv_dateinserted.
  CONCATENATE sy-datum
              sy-uzeit
          INTO gv_dateinserted.
  CLEAR: w_sortc, w_sort.
  SORT gt_h1_mat BY brand gend prodh extwg matnr.
  LOOP AT gt_h1_mat INTO gs_h1_mat.
    CLEAR: w_sortc, gs_deptstyle,gv_matnr,wa_mlist.
    gv_seq = gv_seq + 1.
    w_sort = w_sort + 1.
    w_sortc = w_sort.
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_OUTPUT'
      EXPORTING
        input  = gs_h1_mat-matnr
      IMPORTING
        output = gv_matnr.
    gs_deptstyle-mandt = sy-mandt.
    gs_deptstyle-techsite = p_tsite.
    gs_deptstyle-seq = gv_seq.
    gs_deptstyle-id = gv_seq.
    gs_deptstyle-localeid = gc_localeid_en.
    CONCATENATE gs_h1_mat-brand gs_h1_mat-gend gs_h1_mat-prodh gs_h1_mat-extwg
                gv_matnr
            INTO  gs_deptstyle-hostref.
    gs_deptstyle-zzdeptstyle = gs_deptstyle-departmentstyle = gs_deptstyle-hostref.
    gs_deptstyle-website = gs_techsite-pr_website.
    CONCATENATE gs_h1_mat-brand gs_h1_mat-gend gs_h1_mat-prodh gs_h1_mat-extwg INTO gs_deptstyle-department.
    gs_deptstyle-style = gv_matnr.
    WRITE: w_sortc TO gs_deptstyle-sortorder LEFT-JUSTIFIED.
    gs_deptstyle-dateinserted = gv_dateinserted.
    APPEND gs_deptstyle TO gt_deptstyle.
  ENDLOOP.

  LOOP AT gt_h2_mat INTO gs_h2_mat.
    CLEAR: w_sortc, gs_deptstyle,gv_matnr,wa_mlist.
    gv_seq = gv_seq + 1.
    w_sort = w_sort + 1.
    w_sortc = w_sort.
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_OUTPUT'
      EXPORTING
        input  = gs_h2_mat-matnr
      IMPORTING
        output = gv_matnr.
    gs_deptstyle-mandt = sy-mandt.
    gs_deptstyle-techsite = p_tsite.
    gs_deptstyle-seq = gv_seq.
    gs_deptstyle-id = gv_seq.
    gs_deptstyle-localeid = gc_localeid_en.
    CONCATENATE gs_h2_mat-brand gs_h2_mat-gend gs_h2_mat-matkl gs_h2_mat-prodh gs_h2_mat-extwg
                gv_matnr
            INTO  gs_deptstyle-hostref.
    gs_deptstyle-zzdeptstyle = gs_deptstyle-departmentstyle = gs_deptstyle-hostref.
    gs_deptstyle-website = gs_techsite-pr_website.
    CONCATENATE gs_h2_mat-brand gs_h2_mat-gend gs_h2_mat-matkl gs_h2_mat-prodh gs_h2_mat-extwg INTO gs_deptstyle-department.
    gs_deptstyle-style = gv_matnr.
    WRITE: w_sortc TO gs_deptstyle-sortorder LEFT-JUSTIFIED.
    gs_deptstyle-dateinserted = gv_dateinserted.
