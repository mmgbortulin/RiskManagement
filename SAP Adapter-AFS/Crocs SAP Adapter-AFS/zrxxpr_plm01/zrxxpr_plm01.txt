* DESCRIPTION:  Write automatic entries to CODE and CODES tables

*// local declaration
* SALES REP LIST
  DATA:  BEGIN OF ls_replist,
           rep   TYPE kunnr,
           vkorg TYPE vkorg,
*{   INSERT         DV2K907927                                        3
           vtweg TYPE vtweg,
*}   INSERT
           spart TYPE spart,
           name  TYPE name1,
         END OF ls_replist.

  DATA:  lt_replist LIKE STANDARD TABLE OF ls_replist
                    WITH NON-UNIQUE DEFAULT KEY
                    INITIAL SIZE 50.

* FIELDS
  DATA:  lv_nextyear TYPE datum,
         lv_soldto   TYPE kunnr,       "SOLD TO
         lv_shipto   TYPE kunnr,       "Ship To
         lv_rep      TYPE kunnr,          "Sales Rep
         lv_tabix    TYPE sytabix,      "table index
         lv_cnt      TYPE sydbcnt.        "RECORD COUNTER

*// Begin Processing
  CLEAR:  gs_kode,
          gt_kode[],
          gs_kodes,
          gt_kodes[].

  CLEAR:  gv_seq,
          gv_dateinserted.
  CONCATENATE sy-datum
              sy-uzeit
          INTO gv_dateinserted.

* SET DATE FOR NEXT YEAR
  lv_nextyear = sy-datum + 365.

  SORT gt_soldtolist BY kunnr vkorg vtweg spart.
  DELETE ADJACENT DUPLICATES FROM gt_soldtolist
                             COMPARING kunnr vkorg.

* determine soldto customer's sales reps
  LOOP AT gt_soldtolist ASSIGNING <soldtolist>.
    CLEAR:  lv_soldto,
            ls_replist.
    PERFORM transform_alpha_out USING <soldtolist>-kunnr
                             CHANGING lv_soldto.

*-- Get Partners of Soldto customer
    REFRESH:  gt_knvp.
    SELECT  kunnr vkorg vtweg
            spart parvw parza
            kunn2
        FROM knvp INTO TABLE gt_knvp
        WHERE kunnr = <soldtolist>-kunnr   AND
              vkorg = <soldtolist>-vkorg   AND
              vtweg = <soldtolist>-vtweg   AND
              spart = <soldtolist>-spart   AND
*{   REPLACE        DV2K907927                                       12
*\              PARVW = 'XS'.             "XS: Sales Representative
              parvw = 'ZA'.             "ZS: Sales Representative
*}   REPLACE
    IF sy-subrc = 0.
      LOOP AT gts_repcode ASSIGNING <repcode>
                          WHERE kunag = <soldtolist>-kunnr.
        LOOP AT gt_knvp ASSIGNING <knvp>.
          CLEAR:  lv_rep,
                  lv_shipto,
                  gs_kodes.
          PERFORM transform_alpha_out USING <knvp>-kunn2
                                   CHANGING lv_rep.
          PERFORM transform_alpha_out USING <repcode>-kunwe
                                   CHANGING lv_shipto.
          gs_kodes-mandt = sy-mandt.
          gs_kodes-techsite = gs_techsite-website.
          gs_kodes-salesorg = <soldtolist>-vkorg.
*{   INSERT         DV2K907927                                        2
          gs_kodes-distchnl = <soldtolist>-vtweg.

*}   INSERT
          gs_kodes-division = <soldtolist>-spart.
          gs_kodes-keyword = 'SalesRep'.
          gs_kodes-code = lv_rep.
          CONCATENATE lv_shipto
                      lv_rep
                  INTO gs_kodes-codes SEPARATED BY gv_delimiter.
          gs_kodes-bindingkey = <repcode>-soldtoshipto.
          gs_kodes-qualf = space.
          gs_kodes-localeid = gc_localeid_en.
          gs_kodes-parenttable = 'ShipTo'.
          CONCATENATE sy-datum
                      gc_timestart
                  INTO gs_kodes-dateeffective.
          CONCATENATE lv_nextyear
                      gc_timefinish
                  INTO gs_kodes-dateexpiration.
          gs_kodes-sortorder = space.
          gs_kodes-selected = 0.
          gs_kodes-protected = space.
          APPEND gs_kodes TO gt_kodes.
*------ COLLECT UNIQUE REP LIST
          ls_replist-rep = <knvp>-kunn2.
          ls_replist-vkorg = <knvp>-vkorg.
*{   INSERT         DV2K907927                                        4
          ls_replist-vtweg = <knvp>-vtweg.
*}   INSERT
          ls_replist-spart = <knvp>-spart.
          COLLECT ls_replist INTO lt_replist.
        ENDLOOP.   "AT GT_KNVP ASSIGNING <KNVP>
      ENDLOOP.   "
    ENDIF.   "sy-subrc = 0
  ENDLOOP.   "AT GT_SOLDTOLIST assigning <SOLDTOLIST>

* KODES: sort and remove duplicate records
  SORT gt_kodes BY mandt techsite salesorg
*{   REPLACE        DV2K907927                                        1
*\                   DIVISION KEYWORD CODE
                   distchnl division keyword code
*}   REPLACE
                   codes qualf localeid.
  DELETE ADJACENT DUPLICATES FROM gt_kodes
                             COMPARING mandt techsite salesorg
*{   REPLACE        DV2K907927                                       11
*\                                       DIVISION KEYWORD CODE
                                       distchnl division keyword code
*}   REPLACE
                                       codes qualf localeid.

* Prepare and Add KODES Records
*{   REPLACE        DV2K912096                                       13
*\  delete from ztxx_pr_kodes
*\         where techsite = gs_techsite-website   and
*\               salesorg in s_vkorg              and
*\               keyword = 'SalesRep'             and
*\               protected = space.
*\  insert ztxx_pr_kodes from table gt_kodes
*\                       accepting duplicate keys.
  MODIFY ztxx_pr_kodes FROM TABLE gt_kodes.
*}   REPLACE
  COMMIT WORK.


* Begin KODE table
*{   REPLACE        DV2K907927                                        5
*\  sort lt_replist by rep vkorg SPART.
  SORT lt_replist BY rep vkorg vtweg spart.
*}   REPLACE
  LOOP AT lt_replist ASSIGNING <replist>.
    CLEAR: gs_kode,
           lv_rep.
    gs_kode-mandt = sy-mandt.
    gs_kode-techsite = gs_techsite-website.
    gs_kode-salesorg = <replist>-vkorg.
*{   INSERT         DV2K907927                                        6
    gs_kode-distchnl = <replist>-vtweg.
*}   INSERT
    gs_kode-division = <replist>-spart.
    gs_kode-keyword = 'SalesRep'.
    PERFORM transform_alpha_out USING <replist>-rep
