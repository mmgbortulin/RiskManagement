FUNCTION Z_RFC_PR_HDI_INVOICE_GET.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(I_WEBSITE) TYPE  ZPLUMRIVER_WEBSITE
*"     VALUE(I_VKORG) TYPE  VKORG DEFAULT '%'
*"     VALUE(I_VTWEG) TYPE  VTWEG DEFAULT '%'
*"     VALUE(I_SPART) TYPE  SPART DEFAULT '%'
*"     VALUE(I_ROWS) TYPE  SYTABIX OPTIONAL
*"  TABLES
*"      TA_HEADER STRUCTURE  ZSXX_PR_INVOICE
*"      TA_LINES STRUCTURE  ZSXX_PR_INVOICELINE
*"      TA_RETURN STRUCTURE  BAPIRET1
*"----------------------------------------------------------------------
*=======================================================================
* ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
*  PlumRiver LLC
*  Version 2.0.00
*  Delivered:  April 5, 2011
* ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
*=======================================================================

*//  Local Declarations

*//  Begin Processing
* Initialize tables and set values for parameters
  PERFORM INITIALIZE_GLOBALS_TOP.
* Get technical (SAP) websites
  PERFORM GET_TECHSITES TABLES GT_TECHSITE
                               TA_RETURN
                        using  I_WEBSITE
                     CHANGING  GV_RC.
  IF GV_RC is initial.
    GV_ROWS = i_rows.
*-- Get Invoice HEADER data
    Select *
        from ZTXX_PR_INVOICE  up to GV_ROWS rows
        into corresponding fields of table TA_HEADER
        for all entries in GT_TECHSITE
        where techsite = GT_TECHSITE-WEBSITE   and
              vkorg LIKE i_vkorg    and
              vtweg LIKE i_vtweg    and
              spart LIKE i_spart.
    if sy-subrc = 0.
*---- Successful retrieval of records (Invoice Lines).
      gv_msgid = 'ZPLUMRIVER'.
      write: 'ZTXX_PR_INVOICE' to gv_msgv1 left-justified.
      write: sy-dbcnt to gv_msgv2 left-justified.
      perform SET_RETURN_MESSAGE using 'S' gv_msgid '004'
                                       gv_msgv1 gv_msgv2 space space
                            changing   gs_ret1.
      append gs_ret1 to ta_return.
*---- Get Invoice LINES data
      Select *
          from ZTXX_PR_INVLINE  up to GV_ROWS rows
          into corresponding fields of table TA_LINES
          for all entries in GT_TECHSITE
          where techsite = GT_TECHSITE-WEBSITE   and
                vkorg LIKE i_vkorg    and
                vtweg LIKE i_vtweg    and
                spart LIKE i_spart.
      if sy-subrc = 0.
*------ successful retrieval of records (Invoice Lines).
        gv_msgid = 'ZPLUMRIVER'.
        write: 'ZTXX_PR_INVLINE' to gv_msgv1 left-justified.
        write: sy-dbcnt to gv_msgv2 left-justified.
        perform SET_RETURN_MESSAGE using 'S' gv_msgid '004'
                                         gv_msgv1 gv_msgv2 space space
                              changing   gs_ret1.
        append gs_ret1 to ta_return.
      else.
*------ no records returned (Line Items).
        gv_msgid = 'ZPLUMRIVER'.
        write: 'ZTXX_PR_INVLINE' to gv_msgv1 left-justified.
        perform SET_RETURN_MESSAGE using 'E' gv_msgid '098'
                                         gv_msgv1 space space space
                              changing   gs_ret1.
        append gs_ret1 to ta_return.
*------ Don't return any table values (since no line items)
        REFRESH: TA_HEADER,
                 TA_LINES.
      endif.
    else.
*---- no records returned (Header).
      gv_msgid = 'ZPLUMRIVER'.
      write: 'ZTXX_PR_INVOICE' to gv_msgv1 left-justified.
      perform SET_RETURN_MESSAGE using 'E' gv_msgid '098'
                                       gv_msgv1 space space space
                            changing   gs_ret1.
      append gs_ret1 to ta_return.
    endif.
  ENDIF.   "GV_RC is initial

ENDFUNCTION.
