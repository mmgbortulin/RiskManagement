FORM EVENT_RAISE .
* DESCRIPTION:  Use SAP function to raise processing event

*// Local declaration
  DATA:  LV_RC(1) type N.

*// Begin processing
* Note:  the following function has been flagged obsolete by SAP.
*        Nevertheless, for backward compatibility, we will continue
*        to use the function until actually removed.  At which time,
*        SAP recommends using Class Method, cl_batch_event=>raise.
  CALL FUNCTION 'BP_EVENT_RAISE'
    EXPORTING
      EVENTID                      = P_EVENT
      EVENTPARM                    = P_PARAM
*     TARGET_INSTANCE              = ' '
*     TARGET_MODE                  = ' '
    EXCEPTIONS
      BAD_EVENTID                  = 1
      EVENTID_DOES_NOT_EXIST       = 2
      EVENTID_MISSING              = 3
      RAISE_FAILED                 = 4
      OTHERS                       = 5.
  IF SY-SUBRC = 0.
    MESSAGE I900 WITH P_EVENT
                      P_PARAM.
  ELSE.
    LV_RC = sy-subrc.
    MESSAGE W901 WITH P_EVENT
                      P_PARAM
                      LV_RC.
  ENDIF.

ENDFORM.                    " EVENT_RAISE
