FORM delete_db_rows USING    us_tablekey LIKE gs_tablekey
                             us_tablename TYPE c
                    CHANGING ch_rc TYPE sysubrc.

* LOCAL DECLARATION
  DATA: lv_mandt TYPE mandt.

  CLEAR:  ch_rc.
  DELETE FROM (us_tablename)
         WHERE techsite = us_tablekey-techsite.
  ch_rc = sy-subrc.
  IF ch_rc <> 0.
*-- Did deletion fail because there is no record(s) to delete??
*-- Check for a record... return a successful code (0) if none found
    SELECT SINGLE mandt
            INTO lv_mandt
            FROM (us_tablename)
            WHERE techsite = us_tablekey-techsite.
    IF sy-subrc <> 0.
*---- no records found, new records can be inserted (later)... success.
      ch_rc = 0.
    ENDIF.
  ENDIF.

ENDFORM.                    " DELETE_DB_ROWS
