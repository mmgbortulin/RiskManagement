FORM GET_MATERIAL_DESCRIPTION  USING    US_MATNR TYPE MATNR
                                        US_LOCALEID TYPE ZPLUMRIVER_LID
                               CHANGING CH_NAME type ZPLUMRIVER_URLNAME.
* DESCRIPTION:  GET MATERIAL DESCRIPTION

*// LOCAL DECLARATION
  DATA:  LV_SPRAS TYPE SPRAS.

* LocaleID to SAP language key translation table
  STATICS: BEGIN OF LS_LOCALEID,
             LOCALEID TYPE ZTXX_PR_LOCALEID-LOCALEID,
             SPRAS TYPE ZTXX_PR_LOCALEID-SPRAS,
           END OF LS_LOCALEID.

  STATICS:  LT_LOCALEID LIKE HASHED TABLE OF LS_LOCALEID
                        WITH UNIQUE KEY LOCALEID
                        INITIAL SIZE 10.

*// BEGIN PROCESSING
  IF LT_LOCALEID[] is initial.
*-- RETRIEVE ALL LOCALEIDs.
    SELECT LOCALEID SPRAS
        FROM ZTXX_PR_LOCALEID
        INTO LS_LOCALEID.
      INSERT LS_LOCALEID INTO TABLE LT_LOCALEID.
    ENDSELECT.
  ENDIF.   "LT_LOCALEID[] is initial
* GET SAP language key for localeID
  IF LS_LOCALEID-LOCALEID <> US_LOCALEID.
    READ TABLE LT_LOCALEID INTO LS_LOCALEID
                           WITH TABLE KEY LOCALEID = US_LOCALEID.
    IF SY-SUBRC <> 0.
      MESSAGE E398(00) WITH 'LocaleID'
                            US_LOCALEID
                            'is NOT supported.'
                            'Please maintain.'.
    ENDIF.
  ENDIF.
* Get material text
  clear:  CH_NAME.
  select single maktx
           from MAKT into CH_NAME
           where matnr = us_matnr    and
                 spras = LS_localeid-spras.
  if sy-subrc <> 0     AND
     LS_localeid-spras <> 'E'.         "not English
*-- language text not found... try finding English text
    select single maktx
             from MAKT into CH_NAME
             where matnr = us_matnr    and
                   spras = 'E'.              "English
  endif.   "sy-subrc <> 0...

ENDFORM.                    " GET_MATERIAL_DESCRIPTION

*&---------------------------------------------------------------------*
FORM DELETE_DB_ROWS USING    US_site type ZPLUMRIVER_WEBSITE_TECHNICAL
                             US_tablename type C
                    CHANGING CH_RC type sysubrc.
*  DESCRIPTION: Delete table rows before subsequent update.
*               CH_RC = 0:   Successful
*               CH_RC <> 0:  Failure

* LOCAL DECLARATION
  DATA: LV_MANDT TYPE MANDT.


  clear:  ch_rc.
  delete from (US_TABLENAME)
         WHERE TECHSITE = US_site.
  CH_RC = SY-SUBRC.
  IF CH_RC <> 0.
*-- Did deletion fail because there is no record(s) to delete??
*-- Check for a record... return a successful code (0) if none found
    SELECT SINGLE MANDT
            INTO LV_MANDT
            FROM (US_TABLENAME)
            WHERE TECHSITE = US_SITE.
    IF sy-subrc <> 0.
*---- no records found, new records can be inserted (later)... success.
      CH_RC = 0.
    ENDIF.
  ENDIF.

ENDFORM.                    " DELETE_DB_ROWS
