        lv_string  TYPE string,
        lv_string2 TYPE string.

* Begin processing
  SEARCH ch_text FOR us_find.
  IF sy-subrc = 0.
    CLEAR: lv_string,
           lv_string2.
    lv_string = ch_text.
*    describe field us_find length lv_findlen.
    lv_findlen = strlen( us_find ).
    DO.
      SEARCH lv_string FOR us_find.
      IF sy-subrc = 0.
        lv_len = sy-fdpos + lv_findlen.
        CONCATENATE lv_string2
                    lv_string(sy-fdpos)
                    us_replace
               INTO lv_string2.
        SHIFT lv_string LEFT BY lv_len PLACES.
      ELSE.
        EXIT.
      ENDIF.
    ENDDO.
*-- nothing left to replace...build rest of the string
    CONCATENATE lv_string2
                lv_string
           INTO lv_string2.
    ch_text = lv_string2.
  ENDIF.

ENDFORM.                    " find_and_replace

        gs_orderline-discountpercent = gs_orderline-discountpercent  *
                                       -1.
        APPEND gs_orderline TO gt_orderline_tmp.
      ENDIF.
    ENDLOOP.   "AT GTS_VBEP assigning <VBEP>
*-- if NO Line items exist then skip to next order without saving
    IF gt_orderline_tmp[] IS INITIAL.
      CONTINUE.
    ENDIF.   "GT_ORDERLINE_TMP[] is initial
*-- Complete and fill GS_Order before appending to GT_Order
    PERFORM fill_order_header USING <vbak>
                                    gs_ord01
                                    gx_open
                           CHANGING gs_order.
*-- ALL ORDERS are Added to Order Acknowledgement table (OPEN and
*-- CLOSED).
    APPEND gs_order TO gt_orderack.
    IF p_ordhdr IS INITIAL.
      LOOP AT gt_orderline_tmp ASSIGNING <line>.
        CLEAR: gs_orderackline.
        MOVE-CORRESPONDING <line> TO gs_orderackline.
        APPEND gs_orderackline TO gt_orderackline.
      ENDLOOP.   "AT GT_ORDERLINE_TMP ASSIGNING <LINE>
    ENDIF.
*-- ORDER IS ALSO OPEN??
*    IF NOT gx_open IS INITIAL.
*---- update header amounts determined from line items.
    CLEAR:  gs_order-totalunits,
            gs_order-subtotalwholesal,
            gs_order-subtotalnet.
    LOOP AT gt_orderline_tmp ASSIGNING <line>.
      lv_orderqty = <line>-qtyordered.
      IF lv_orderqty > '0.0'.
*-------- Total OPEN Units
        gs_order-totalunits = gs_order-totalunits  +
                              <line>-qtybalance.
*-------- Subtotal Wholesale (Open ONLY)
        <line>-subtotalwholesal = <line>-subtotalwholesal *
                                  <line>-qtybalance / lv_orderqty.
        gs_order-subtotalwholesal = gs_order-subtotalwholesal  +
                                   <line>-subtotalwholesal.
*-------- Subtotal Net (Open only)
        <line>-subtotalnet = <line>-subtotalnet *
                             <line>-qtybalance / lv_orderqty.
        gs_order-subtotalnet = gs_order-subtotalnet  +
                              <line>-subtotalnet.
      ENDIF.   "<LINE>-QTYORDERED > '0.0'
*------ Now that the line valuation is complete for Open Orders... for
*       display on the website, field QTYBALANCE must reflect QTYOPEN.
*       REASON:  QTYBALANCE displays the OPEN Qty on the website, NOT
*                field QTYOPEN.
      <line>-qtybalance = <line>-qtyopen.
    ENDLOOP.   "AT GT_ORDERLINE_TMP ASSIGNING <LINE>
    APPEND gs_order TO gt_order.
    IF p_ordhdr IS INITIAL.
      APPEND LINES OF gt_orderline_tmp TO gt_orderline.
    ENDIF.
*    ENDIF.
  ENDLOOP.   "AT GT_VBAK ASSIGNING <VBAK>
ENDFORM.                    " SUB_ORD_HDR_LINE

FORM sub_get_changed_ord CHANGING ch_rc TYPE sy-subrc.
  TYPES: BEGIN OF t_vbak,
           vbeln TYPE vbeln,
         END OF t_vbak,

         BEGIN OF t_ordchg,
           zzorder TYPE zplumriver_orderno,
           ordstat TYPE zplumriver_orderstatus,
         END OF t_ordchg,

         BEGIN OF t_date_inst,
           sign   TYPE char1,
           option TYPE char2,
           low    TYPE zplumriver_dateinserted,
           high   TYPE zplumriver_dateinserted,
         END OF t_date_inst,

         BEGIN OF t_aedat,
           sign   TYPE char1,
           option TYPE char2,
           low    TYPE datum,
           high   TYPE datum,
         END OF t_aedat.

  DATA: it_vbak      TYPE TABLE OF t_vbak,
        it_ordchg    TYPE TABLE OF t_ordchg,
        it_date_inst TYPE TABLE OF t_date_inst,
        it_aedat     TYPE TABLE OF t_aedat.

  DATA: wa_ordchg    TYPE t_ordchg,
        wa_date_inst TYPE t_date_inst.

  DATA: l_date  TYPE zplumriver_dateinserted,
        l_cdat  TYPE char10,
        l_dat   TYPE sy-datum,
        l_mdate TYPE zplumriver_dateordered,
        l_mxdat TYPE zplumriver_dateordered,
        l_tab   TYPE tabname.
  FIELD-SYMBOLS: <fs_vbak>   TYPE t_vbak,
                 <fs_ordchg> TYPE t_ordchg,
                 <fs_aedat>  TYPE t_aedat.

  CLEAR: it_vbak,it_aedat.
  l_tab = 'ZTXX_PR_ORDERHDR'.
  IF p_nomon IS NOT INITIAL.
    CALL FUNCTION 'OIUPR_DATE_SUBTRACT_MONTH'
      EXPORTING
        date       = sy-datum
        months     = p_nomon
      IMPORTING
        date_e     = l_dat
      EXCEPTIONS
        date_error = 1
        OTHERS     = 2.
    IF sy-subrc EQ 0.
      CONCATENATE l_dat gc_timestart INTO l_mdate.
      CONCATENATE sy-datum gc_timefinish INTO l_mxdat.
      CLEAR: wa_date_inst,it_date_inst.
      wa_date_inst-sign = c_i.
      wa_date_inst-option = c_bt.
      wa_date_inst-low = l_mdate.
      wa_date_inst-high = l_mxdat.
      APPEND wa_date_inst TO it_date_inst.
      SELECT zzorder orderstatus FROM ztxx_pr_orderhdr INTO TABLE it_ordchg
      WHERE techsite = p_tsite
        AND vkorg    IN s_vkorg
        AND vtweg    IN s_vtweg
        AND spart    IN s_spart
        AND dateinserted IN it_date_inst[].
      IF sy-subrc EQ 0.
        DELETE it_ordchg WHERE ordstat EQ c_comp.
        LOOP AT it_ordchg ASSIGNING <fs_ordchg>.
          APPEND INITIAL LINE TO it_vbak ASSIGNING <fs_vbak>.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
            EXPORTING
              input  = <fs_ordchg>-zzorder
            IMPORTING
              output = <fs_ordchg>-zzorder.
          <fs_vbak>-vbeln = <fs_ordchg>-zzorder.
        ENDLOOP.
      ENDIF.
    ENDIF.
  ENDIF.
  IF s_aedat[] IS NOT INITIAL.
    CLEAR: it_date_inst.
    LOOP AT s_aedat.
      CLEAR: l_date,l_cdat,wa_date_inst.
      wa_date_inst-sign = s_aedat-sign.
      wa_date_inst-option = s_aedat-option.
      IF s_aedat-low IS NOT INITIAL.
        l_date = s_aedat-low.
        CONCATENATE l_date gc_timestart INTO wa_date_inst-low.
      ENDIF.
      IF s_aedat-high IS NOT INITIAL.
        CONCATENATE l_cdat gc_timefinish INTO wa_date_inst-high.
      ENDIF.
      APPEND wa_date_inst TO it_date_inst.
    ENDLOOP.
    IF s_aedat-low IS NOT INITIAL.
      LOOP AT s_aedat WHERE sign EQ c_i.
        CLEAR: l_date,wa_date_inst.
        l_date = s_aedat-low - 1.
        CONCATENATE l_date gc_timestart INTO wa_date_inst-low.
        CONCATENATE s_aedat-low gc_timefinish INTO wa_date_inst-high.
        wa_date_inst-sign = c_i.
        wa_date_inst-option = c_bt.
        APPEND wa_date_inst TO it_date_inst.
      ENDLOOP.
    ENDIF.
    CLEAR: it_ordchg.
    SELECT zzorder orderstatus FROM ztxx_pr_orderhdr INTO TABLE it_ordchg
    WHERE techsite = p_tsite
      AND vkorg    IN s_vkorg
      AND vtweg    IN s_vtweg
      AND spart    IN s_spart
      AND dateinserted IN it_date_inst[].
    IF sy-subrc EQ 0.
      DELETE it_ordchg WHERE ordstat EQ c_comp.
      LOOP AT it_ordchg ASSIGNING <fs_ordchg>.
        APPEND INITIAL LINE TO it_vbak ASSIGNING <fs_vbak>.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING
            input  = <fs_ordchg>-zzorder
          IMPORTING
            output = <fs_ordchg>-zzorder.
        <fs_vbak>-vbeln = <fs_ordchg>-zzorder.
      ENDLOOP.
    ENDIF.
  ENDIF.
  CLEAR: l_date.
  SELECT SINGLE MAX( dateinserted ) FROM (l_tab) INTO l_date.
  IF sy-subrc EQ 0.
    APPEND INITIAL LINE TO it_aedat ASSIGNING <fs_aedat>.
    <fs_aedat>-sign = c_i.
    <fs_aedat>-option = c_bt.
    <fs_aedat>-low = l_date+0(8).
    <fs_aedat>-high = sy-datum.
    IF NOT it_aedat[] IS INITIAL.
      SELECT vbeln FROM vbak APPENDING TABLE it_vbak
      WHERE erdat IN it_aedat[]
        AND auart IN s_auart
        AND vkorg IN s_vkorg
        AND vtweg IN s_vtweg
        AND spart IN s_spart.
      ch_rc = sy-subrc.
    ENDIF.
  ENDIF.
  IF it_vbak[] IS NOT INITIAL.
    SORT it_vbak BY vbeln.
    DELETE ADJACENT DUPLICATES FROM it_vbak COMPARING vbeln.
    SELECT *
       FROM vbak INTO TABLE gt_vbak FOR ALL ENTRIES IN it_vbak
       WHERE vbeln EQ it_vbak-vbeln    AND
             auart IN s_auart    AND
             vbtyp IN s_vbtyp    AND
             vkorg IN s_vkorg    AND
             vtweg IN s_vtweg    AND
             spart IN s_spart    AND
             kunnr IN s_kunag    AND
             audat IN s_audat    AND
             vdatu IN s_vdatu.
    ch_rc = sy-subrc.
  ELSE.
    ch_rc = 4.
  ENDIF.
ENDFORM.                    " SUB_GET_CHANGED_ORD

FORM sub_get_missed_recs  CHANGING ch_rc TYPE sy-subrc.
  TYPES: BEGIN OF t_vbak,
           vbeln TYPE vbeln,
         END OF t_vbak,

         BEGIN OF t_ord,
           sign   TYPE char1,
           option TYPE char2,
           low    TYPE vbeln,
           high   TYPE vbeln,
         END OF t_ord,

         BEGIN OF t_ordchg,
           zzorder TYPE vbeln,
           ordstat TYPE zplumriver_orderstatus,
         END OF t_ordchg,

         BEGIN OF t_date_inst,
           sign   TYPE char1,
           option TYPE char2,
           low    TYPE zplumriver_dateinserted,
           high   TYPE zplumriver_dateinserted,
         END OF t_date_inst,

         BEGIN OF t_aedat,
           sign   TYPE char1,
           option TYPE char2,
           low    TYPE datum,
           high   TYPE datum,
         END OF t_aedat.

  TYPES: BEGIN OF t_knvv,
           kunnr TYPE kunnr,
         END OF t_knvv.

  DATA: it_knvv TYPE TABLE OF t_knvv.

  DATA: it_vbak      TYPE TABLE OF t_vbak,
        it_ordno     TYPE TABLE OF t_vbak,
        it_ord       TYPE TABLE OF t_ord,
        it_ordchg    TYPE TABLE OF t_ordchg,
        it_ordc      TYPE TABLE OF t_ordchg,
        it_date_inst TYPE TABLE OF t_date_inst,
        it_date      TYPE TABLE OF t_aedat.

  DATA: wa_ordchg    TYPE t_ordchg,
        wa_date      TYPE t_aedat,
        wa_date_inst TYPE t_date_inst.

  DATA: l_date  TYPE zplumriver_dateinserted,
        l_cdat  TYPE char10,
        l_dat   TYPE sy-datum,
        l_mdate TYPE zplumriver_dateordered,
        l_mxdat TYPE zplumriver_dateordered,
        l_tab   TYPE tabname.
  FIELD-SYMBOLS: <fs_vbak>   TYPE t_vbak,
                 <fs_ordno>  TYPE t_vbak,
                 <fs_ord>    TYPE t_ord,
                 <fs_ordchg> TYPE t_ordchg.

  CLEAR: it_vbak,it_date,it_date_inst,it_ordchg.
  l_tab = 'ZTXX_PR_ORDERHDR'.
  IF p_nomon IS NOT INITIAL.
    CALL FUNCTION 'OIUPR_DATE_SUBTRACT_MONTH'
      EXPORTING
        date       = sy-datum
        months     = p_nomon
      IMPORTING
        date_e     = l_dat
      EXCEPTIONS
        date_error = 1
        OTHERS     = 2.
    IF sy-subrc EQ 0.
      CONCATENATE l_dat gc_timestart INTO l_mdate.
      CONCATENATE sy-datum gc_timefinish INTO l_mxdat.
      CLEAR: wa_date_inst,it_date_inst.
      wa_date-sign = wa_date_inst-sign = c_i.
      wa_date-option = wa_date_inst-option = c_bt.
      wa_date_inst-low = l_mdate.
      wa_date_inst-high = l_mxdat.
      wa_date-low = l_dat.
      wa_date-high = sy-datum.
      APPEND wa_date_inst TO it_date_inst.
      APPEND wa_date TO it_date.
      IF NOT s_erdat[] IS INITIAL.
        APPEND LINES OF s_erdat[] TO it_date[].
      ENDIF.
      SELECT zzorder orderstatus FROM ztxx_pr_orderhdr INTO TABLE it_ordchg
      WHERE techsite = p_tsite
        AND vkorg    IN s_vkorg
        AND vtweg    IN s_vtweg
        AND spart    IN s_spart
        AND dateinserted IN it_date_inst[].
      IF sy-subrc EQ 0.
        it_ordc[] = it_ordchg[].
        DELETE it_ordchg WHERE ordstat EQ c_comp.
        LOOP AT it_ordchg ASSIGNING <fs_ordchg>.
          APPEND INITIAL LINE TO it_vbak ASSIGNING <fs_vbak>.
          <fs_vbak>-vbeln = <fs_ordchg>-zzorder.
        ENDLOOP.
        LOOP AT it_ordc ASSIGNING <fs_ordchg>.
          APPEND INITIAL LINE TO it_ord ASSIGNING <fs_ord>.
          <fs_ord>-sign = c_i.
          <fs_ord>-option = c_eq.
          <fs_ord>-low = <fs_ordchg>-zzorder.
        ENDLOOP.
        SELECT kunnr FROM knvv INTO TABLE it_knvv WHERE kunnr IN s_kunag
                                                    AND vkorg IN s_vkorg
                                                    AND vtweg IN s_vtweg
                                                    AND spart IN s_spart
                                                    AND kdgrp IN s_kdgrp.
        IF NOT it_knvv[] IS INITIAL.
          SELECT vbeln
             FROM vbak INTO TABLE it_ordno FOR ALL ENTRIES IN it_knvv
             WHERE vbeln IN s_vbeln    AND
                   auart IN s_auart    AND
                   vbtyp IN s_vbtyp    AND
                   vkorg IN s_vkorg    AND
                   vtweg IN s_vtweg    AND
                   spart IN s_spart    AND
                   kunnr = it_knvv-kunnr AND
                   audat IN s_audat   AND
                   erdat IN it_date[] AND
                   vdatu IN s_vdatu.
          IF sy-subrc EQ 0.
            SORT it_ordno.
            IF it_ord[] IS NOT INITIAL.
              DELETE it_ordno WHERE vbeln IN it_ord[].
            ENDIF.
            LOOP AT it_ordno ASSIGNING <fs_ordno>.
              APPEND INITIAL LINE TO it_vbak ASSIGNING <fs_vbak>.
              <fs_vbak>-vbeln = <fs_ordno>-vbeln.
            ENDLOOP.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
  ELSE.
    IF NOT s_erdat[] IS INITIAL.
      APPEND LINES OF s_erdat[] TO it_date[].
    ENDIF.
    SELECT zzorder orderstatus FROM ztxx_pr_orderhdr INTO TABLE it_ordchg
    WHERE techsite = p_tsite
      AND vkorg    IN s_vkorg
      AND vtweg    IN s_vtweg
      AND spart    IN s_spart.
    IF sy-subrc EQ 0.
      it_ordc[] = it_ordchg[].
      DELETE it_ordchg WHERE ordstat EQ c_comp.
      LOOP AT it_ordchg ASSIGNING <fs_ordchg>.
        APPEND INITIAL LINE TO it_vbak ASSIGNING <fs_vbak>.
        <fs_vbak>-vbeln = <fs_ordchg>-zzorder.
      ENDLOOP.
      LOOP AT it_ordc ASSIGNING <fs_ordchg>.
        APPEND INITIAL LINE TO it_ord ASSIGNING <fs_ord>.
        <fs_ord>-sign = c_i.
        <fs_ord>-option = c_eq.
        <fs_ord>-low = <fs_ordchg>-zzorder.
      ENDLOOP.
      SELECT kunnr FROM knvv INTO TABLE it_knvv WHERE kunnr IN s_kunag
                                                  AND vkorg IN s_vkorg
                                                  AND vtweg IN s_vtweg
                                                  AND spart IN s_spart
                                                  AND kdgrp IN s_kdgrp.
      IF NOT it_knvv[] IS INITIAL.
        SELECT vbeln
           FROM vbak INTO TABLE it_ordno FOR ALL ENTRIES IN it_knvv
           WHERE vbeln IN s_vbeln    AND
                 auart IN s_auart    AND
                 vbtyp IN s_vbtyp    AND
                 vkorg IN s_vkorg    AND
                 vtweg IN s_vtweg    AND
                 spart IN s_spart    AND
                 kunnr = it_knvv-kunnr AND
                 audat IN s_audat   AND
                 erdat IN it_date[] AND
                 vdatu IN s_vdatu.
        IF sy-subrc EQ 0.
          SORT it_ordno.
          IF it_ord[] IS NOT INITIAL.
            DELETE it_ordno WHERE vbeln IN it_ord[].
          ENDIF.
          LOOP AT it_ordno ASSIGNING <fs_ordno>.
            APPEND INITIAL LINE TO it_vbak ASSIGNING <fs_vbak>.
            <fs_vbak>-vbeln = <fs_ordno>-vbeln.
          ENDLOOP.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.
  IF it_vbak[] IS NOT INITIAL.
    SORT it_vbak BY vbeln.
    DELETE ADJACENT DUPLICATES FROM it_vbak COMPARING vbeln.
    SELECT *
       FROM vbak APPENDING TABLE gt_vbak FOR ALL ENTRIES IN it_vbak
       WHERE vbeln EQ it_vbak-vbeln    AND
             auart IN s_auart    AND
             vbtyp IN s_vbtyp    AND
             vkorg IN s_vkorg    AND
             vtweg IN s_vtweg    AND
             spart IN s_spart    AND
             kunnr IN s_kunag    AND
             audat IN s_audat    AND
             vdatu IN s_vdatu.
    ch_rc = sy-subrc.
  ELSE.
    ch_rc = 4.
  ENDIF.
ENDFORM.                    " SUB_GET_MISSED_RECS
