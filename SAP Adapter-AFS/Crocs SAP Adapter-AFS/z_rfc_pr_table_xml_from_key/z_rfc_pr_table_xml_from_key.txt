FUNCTION z_rfc_pr_table_xml_from_key.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_TABLECODE) TYPE  ZPLUMRIVER_HDITABNAME
*"  EXPORTING
*"     VALUE(E_TABLECODE) TYPE  ZPLUMRIVER_HDITABNAME
*"     VALUE(E_COUNT) TYPE  CHAR10
*"  TABLES
*"      TA_TABKEY STRUCTURE  ZSXX_PR_TABLEKEY OPTIONAL
*"      TA_XMLTEXT STRUCTURE  ZSXX_PR_TEXTROW OPTIONAL
*"      TA_RETURN STRUCTURE  BAPIRET1 OPTIONAL
*"----------------------------------------------------------------------
*=======================================================================
* ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
*  PlumRiver LLC
*  Version 2.0.00
*  Delivered:  October 20, 2011
* ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
*=======================================================================


*//  Local Declarations
  DATA:  ls_tabkey TYPE zsxx_pr_tablekey,
         ls_xmltext TYPE zsxx_pr_textrow.

  FIELD-SYMBOLS:  <tabkey>  LIKE LINE OF ta_tabkey,
                  <xmltext> LIKE LINE OF ta_xmltext.

*//  Begin Processing
* Return table code
  e_tablecode = i_tablecode.

* Initialize tables and set values for parameters
  PERFORM initialize_globals_top.

  CLEAR: gv_dbcnt.
*-- determine name of table to be transferred
  PERFORM get_tablename TABLES ta_return
                         USING i_tablecode
                      CHANGING gv_tabname.
  IF NOT gv_tabname IS INITIAL.
*-- Begin data extraction
********************************************
*-- we have a table to process...
*-- First, build HDI fieldname exception/translation list
    SELECT *
        FROM ztxx_pr_fieldmap INTO gs_hdifields
        WHERE tabname = gv_tabname.
      INSERT gs_hdifields INTO TABLE gth_hdifields.
    ENDSELECT.
*-- Now, continue processing
    CLEAR: dref,
           gt_fields.
*-- get table field names
    SELECT  tabname fieldname as4local
            position keyflag rollname
            inttype intlen precfield
            datatype leng decimals
            domname comptype
        FROM dd03l INTO TABLE gt_fields
        WHERE tabname = gv_tabname   AND
              as4local = 'A'.                       "active
    IF sy-subrc = 0.
*---- table fields retrieved... begin processing.
      SORT gt_fields BY position.
*---- remove any fields that are key table fields or don't have
*---- a data element as a reference data type
      DELETE gt_fields WHERE comptype <> 'E'   OR   "not data element
                             keyflag = 'X'.         "primary key
*---- prepare administrative fields.
      CLEAR: gv_line,
             gv_tabix.
      LOOP AT gt_fields ASSIGNING <fields>.
        gv_tabix = gv_tabix + 1.
*------ Check for HDI fieldname exceptions (that don't match)
        READ TABLE gth_hdifields ASSIGNING <hdi>
                      WITH TABLE KEY tabname = <fields>-tabname
                                     fieldname = <fields>-fieldname.
        IF sy-subrc = 0.
*-------- corresponding HDI fieldname does not match SAP DB fieldname
          <fields>-hdifield = <hdi>-hdifield.
          <fields>-xlongtext = <hdi>-xlongtext.
        ELSE.
*-------- HDI fieldname is same in SAP database table
          <fields>-hdifield = <fields>-fieldname.
        ENDIF.
*------ Create OPEN and CLOSE XML tags
        CONCATENATE '<'
                    <fields>-hdifield
                    '>'
                INTO <fields>-otag.
        CONCATENATE '</'
                    <fields>-hdifield
                    '>'
                INTO <fields>-ctag.
      ENDLOOP.  "at gt_fields assigning <fields>
    ELSE.
*---- output message that table fields could not be determined
      gv_msgid = 'ZPLUMRIVER'.
      WRITE: gv_tabname TO gv_msgv1 LEFT-JUSTIFIED.
      PERFORM set_return_message USING 'E' gv_msgid '015'
                                       gv_msgv1 space space space
                            CHANGING   gs_ret1.
      APPEND gs_ret1 TO ta_return.
    ENDIF.   "sy-subrc = 0 [when retrieving table field names]
*-- Now begin retrieving records from the database table
    CREATE DATA dref TYPE (gv_tabname).
    ASSIGN dref->* TO <wa>.
*-- LOOP ON PRIMARY KEY LIST TO BUILD XML TABLE
    LOOP AT ta_tabkey ASSIGNING <tabkey>.
      SELECT *
          FROM (gv_tabname)
          INTO <wa>
          WHERE  techsite = <tabkey>-techsite    AND
                 vkorg = <tabkey>-vkorg          AND
                 vtweg = <tabkey>-vtweg          AND
                 spart = <tabkey>-spart          AND
                 seq = <tabkey>-seq.
        IF sy-subrc = 0.
          CLEAR: gv_line.
          gv_dbcnt = gv_dbcnt + 1.
          LOOP AT gt_fields ASSIGNING <fields>.
            ASSIGN COMPONENT <fields>-fieldname OF STRUCTURE <wa>
                                                TO <fld>.
            IF sy-subrc <> 0.
              UNASSIGN <fld>.
            ENDIF.
*-------- HANDLE LONG TEXTS (> CHAR255) IN THE FOLLOWING CASE STATEMENT.
            IF <fields>-xlongtext = 'X'.
              PERFORM get_sap_longtext  USING  <fld>
                                     CHANGING  gv_fieldstring.
              ASSIGN gv_fieldstring TO <fld>.
              IF sy-subrc <> 0.
                UNASSIGN <fld>.
              ENDIF.
            ENDIF.   "<fields>-xlongtext = 'X'
            IF <fld> IS ASSIGNED.
              IF <fields>-domname = 'ZPLUMRIVER_TIMESTAMP'.
*------------ provide selected date/time fields in HDI timestamp format
*------------ CCYY-MM-DD HH:MM:SS
                CLEAR: gv_hdidate.
                WRITE: <fld> TO gv_hdidate.
                gv_string = gv_hdidate.
              ELSE.
*------------ everything else
                gv_string = <fld>.
              ENDIF.   "<fields>-domname = 'ZPLUMRIVER_TIMESTAMP'
**--------- build tagged field
              IF <fields>-inttype CA 'PFIX'.
*------------ numeric types
                SHIFT gv_string LEFT DELETING LEADING space.
              ENDIF.   "<fields>-inttype ca 'PFIX'
              CONCATENATE gv_line
                          <fields>-otag
                          gv_string
                          <fields>-ctag
                      INTO gv_line.
            ENDIF.    "<fld> is assigned
          ENDLOOP.   "at gt_fields assigning <fields>
*------ transfer data record to return table
          CLEAR: gv_pos,
                 gv_len,
                 gv_strlen,
                 ls_xmltext.
          gv_strlen = NUMOFCHAR( gv_line ).
          ls_xmltext-id = 'NL'.
          WHILE gv_strlen > 0.
            IF gv_strlen > 500.
              gv_len = 500.
            ELSE.
              gv_len = gv_strlen.
            ENDIF.
            ls_xmltext-tline = gv_line+gv_pos(gv_len).
            APPEND ls_xmltext TO ta_xmltext.
            CLEAR: ls_xmltext.
            ls_xmltext-id = '/='.        "subsequent records (continued)
            gv_pos = gv_pos + gv_len.    "new string position
            gv_strlen = gv_strlen - 500. "remaining line characters
          ENDWHILE.   "gv_strlen > 0
        ENDIF.   "
      ENDSELECT.
    ENDLOOP.   "AT TA_TABKEY ASSIGNING <TABKEY>
  ENDIF.   "NOT GV_TABNAME IS INITIAL
* Send final Message to Return TABLE for messages (Success/Failure)
  IF NOT ta_xmltext[] IS INITIAL.
*-- successful retrieval of records.
    gv_msgid = 'ZPLUMRIVER'.
    WRITE: gv_dbcnt TO gv_msgv1 LEFT-JUSTIFIED.
    PERFORM set_return_message USING 'S' gv_msgid '003'
                                     gv_msgv1 space space space
                          CHANGING   gs_ret1.
    APPEND gs_ret1 TO ta_return.
  ELSE.
*---- no records returned.
    gv_msgid = 'ZPLUMRIVER'.
    PERFORM set_return_message USING 'E' gv_msgid '099'
                                     space space space space
                          CHANGING   gs_ret1.
    APPEND gs_ret1 TO ta_return.
  ENDIF.   "NOT TA_XMLTEXT[] is initial

* return record count
  WRITE:  gv_dbcnt TO e_count
                   NO-GROUPING
                   LEFT-JUSTIFIED.

ENDFUNCTION.
