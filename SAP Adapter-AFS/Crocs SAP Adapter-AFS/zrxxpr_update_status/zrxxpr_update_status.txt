*&---------------------------------------------------------------------*
*&      Form  SET_GLOBALS
*&---------------------------------------------------------------------*
FORM set_globals.
* set intial values for global fields and work areas

* initialize variables and structures
  CLEAR:  gs_techsite.

* Get the technical website details
  SELECT SINGLE  website pr_website url
            FROM ztxx_pr_website INTO gs_techsite
            WHERE website = p_tsite.
  IF sy-subrc <> 0.
    CLEAR:  gs_techsite.
    MESSAGE:  e011(zplumriver) WITH p_tsite.
    STOP.   "and go to END-OF-SELECTION
  ENDIF.

ENDFORM.                    " SET_GLOBALS


*&---------------------------------------------------------------------*
*&      Form  SET_NEW_STATUS
*&---------------------------------------------------------------------*
FORM set_new_status.
* DESCRIPTION:  Set new status and update log table

  CLEAR:  gs_hstatus,
          gs_log.

* GET CURRENT STATUS
  SELECT SINGLE *
           FROM ztxx_pr_hstatus
           INTO gs_hstatus
           WHERE techsite = gs_techsite-website.
* BUILD LOG ENTRY (IT DOESN'T MATTER WHETHER ZTXX_PR_HSTATUS RECORD
* WAS FOUND IN PREVIOUS SELECT SINGLE STEP OR NOT).
  gs_log-mandt = sy-mandt.
  gs_log-techsite = gs_techsite-website.
  gs_log-process_date = sy-datum.
  gs_log-process_time = sy-uzeit.
  gs_log-new_status = p_status.
  gs_log-old_status = gs_hstatus-status.
* SET NEW STATUS
  IF gs_log-new_status = gs_log-old_status.
    IF p_log = 'X'.
      INSERT ztxx_pr_lstatus FROM gs_log.
      IF sy-subrc = 0.
        COMMIT WORK.
        MESSAGE s253(zplumriver) WITH gs_log-old_status
                                      gs_hstatus-status.
      ELSE.
        ROLLBACK WORK.
        MESSAGE w254(zplumriver) WITH 'ZTXX_PR_LSTATUS' gs_log-old_status gs_hstatus-status.
      ENDIF.   "SY-SUBRC = 0  [ON INSERT ZTXX_PR_LSTATUS...]
    ELSE.
*---- NO STATUS CHANGE... EXIT WITHOUT UPDATE.
      MESSAGE i250(zplumriver) WITH p_status.
    ENDIF.   "P_LOG = 'X'
  ELSE.
*-- UPDATE WITH NEW STATUS
    gs_hstatus-mandt = gs_log-mandt.
    gs_hstatus-techsite = gs_log-techsite.
    gs_hstatus-status = gs_log-new_status.
    CONCATENATE gs_log-process_date
                gs_log-process_time
             INTO gs_hstatus-dateinserted.
    IF NOT gs_hstatus IS INITIAL.
      MODIFY ztxx_pr_hstatus FROM gs_hstatus.
      IF sy-subrc = 0.
*------ HEADER STATUS MODIFIED (PENDING COMMIT)
        INSERT ztxx_pr_lstatus FROM gs_log.
        IF sy-subrc = 0.
          COMMIT WORK.
          MESSAGE s253(zplumriver) WITH gs_log-old_status
                                        gs_hstatus-status.
        ELSE.
          ROLLBACK WORK.
          MESSAGE w254(zplumriver) WITH 'ZTXX_PR_LSTATUS' gs_log-old_status gs_hstatus-status.
        ENDIF.   "SY-SUBRC = 0  [ON INSERT ZTXX_PR_LSTATUS...]
      ELSE.
*------ STATUS NOT MODIFIED
        MESSAGE w254(zplumriver) WITH 'ZTXX_PR_HSTATUS'  gs_log-old_status gs_hstatus-status..
      ENDIF.
    ENDIF.   "NOT GS_HSTATUS IS INITIAL
  ENDIF.   "GS_LOG-NEW_STATUS = GS_LOG-OLD_STATUS

ENDFORM.                    " SET_NEW_STATUS
