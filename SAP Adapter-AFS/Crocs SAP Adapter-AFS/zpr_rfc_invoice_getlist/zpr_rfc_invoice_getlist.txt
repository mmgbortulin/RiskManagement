FUNCTION zpr_rfc_invoice_getlist.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(WEBSITE) TYPE  ZPLUMRIVER_WEBSITE_TECHNICAL OPTIONAL
*"     VALUE(SOLD_TO) TYPE  KUNNR OPTIONAL
*"     VALUE(FROM_DATE) TYPE  DATUM OPTIONAL
*"     VALUE(TO_DATE) TYPE  DATUM OPTIONAL
*"     VALUE(FROM_INVOICE) TYPE  VBELN OPTIONAL
*"     VALUE(TO_INVOICE) TYPE  VBELN OPTIONAL
*"     VALUE(SALES_ORDER) TYPE  VBELN OPTIONAL
*"     VALUE(SHIP_TO) TYPE  KUNNR OPTIONAL
*"     VALUE(FROM_WEB_ORDER) TYPE  IHREZ OPTIONAL
*"     VALUE(TO_WEB_ORDER) TYPE  IHREZ OPTIONAL
*"     VALUE(CUST_PO) TYPE  BSTKD OPTIONAL
*"     VALUE(FROM_SHIP_DATE) TYPE  DATUM OPTIONAL
*"     VALUE(TO_SHIP_DATE) TYPE  DATUM OPTIONAL
*"     VALUE(FROM_SALES_ORD_DATE) TYPE  DATUM OPTIONAL
*"     VALUE(TO_SALES_ORD_DATE) TYPE  DATUM OPTIONAL
*"     VALUE(STYLE) TYPE  MATNR OPTIONAL
*"     VALUE(TEST) TYPE  FLAG OPTIONAL
*"  EXPORTING
*"     VALUE(ROWCOUNT) TYPE  I
*"  TABLES
*"      INV_HDR STRUCTURE  ZSXX_PR_INVOICE OPTIONAL
*"      DOC_NO STRUCTURE  ZSPR_DOC OPTIONAL
*"      SALES_ORG_IN STRUCTURE  ZSPR_VKORG OPTIONAL
*"      DIST_CHANNEL_IN STRUCTURE  ZSPR_VTWEG OPTIONAL
*"      DIVISION_IN STRUCTURE  ZSPR_SPART OPTIONAL
*"      DOC_TYPE_IN STRUCTURE  ZSPR_FKART OPTIONAL
*"----------------------------------------------------------------------
  DATA: r_vbeln    TYPE RANGE OF vbeln,
        r_auart    TYPE RANGE OF auart,
        r_vbtyp    TYPE RANGE OF vbtyp,
        r_vkorg    TYPE RANGE OF vkorg,
        r_vtweg    TYPE RANGE OF vtweg,
        r_spart    TYPE RANGE OF spart,
        r_kunag    TYPE RANGE OF kunag,
        r_kdgrp    TYPE RANGE OF kdgrp,
        r_audat    TYPE RANGE OF audat,
        r_erdat    TYPE RANGE OF datum,
        r_aedat    TYPE RANGE OF aedat,
        r_vdatu    TYPE RANGE OF datum,
        r_matnr    TYPE RANGE OF matnr,
        r_abgru    TYPE RANGE OF abgru,
        r_fkart    TYPE RANGE OF fkart,
        r_kunnr    TYPE RANGE OF kunnr,
        r_variant  TYPE RANGE OF variant,
        r_webord   TYPE RANGE OF ihrez,
        r_shipdt   TYPE RANGE OF vbak-vdatu,
        r_custpo   TYPE RANGE OF vbrk-bstnk_vf,
        r_style    TYPE RANGE OF matnr,
        r_order    TYPE RANGE OF vbeln,
        r_shipto   TYPE RANGE OF kunnr,
        r_sodate   TYPE RANGE OF datum,
        r_svbtyp   TYPE RANGE OF vbak-vbtyp,
        rl_variant LIKE LINE  OF r_variant,
        rl_webord  LIKE LINE  OF r_webord,
        rl_shipdt  LIKE LINE  OF r_shipdt,
        rl_custpo  LIKE LINE  OF r_custpo,
        rl_style   LIKE LINE  OF r_style,
        rl_order   LIKE LINE  OF r_order,
        rl_shipto  LIKE LINE  OF r_shipto,
        rl_sodate  LIKE LINE  OF r_sodate.
  .

  DATA: BEGIN OF ls_totunit,
          vbeln TYPE vbeln,
          fkimg TYPE fkimg,
        END OF ls_totunit.
  DATA: lt_totunit LIKE TABLE OF ls_totunit.

********************
  DATA: l_uzeit TYPE uzeit.
  l_uzeit = sy-uzeit + 120.
  IF test = 'X'.
    DO.
      IF test = ' '.
        EXIT.
      ENDIF.
      IF sy-uzeit > l_uzeit.
        EXIT.
      ENDIF.
    ENDDO.
  ENDIF.
**************
  PERFORM initialize_globals.

  CONCATENATE sy-datum sy-uzeit INTO gv_dateinserted.

  PERFORM get_website_variants TABLES r_variant
*                                       sales_org_in
                                 USING 'ZPR_RFC_INVOICE_HIST_VARIANT'
                                       website.

  CLEAR rl_variant.
  READ TABLE r_variant INTO rl_variant INDEX 1.


  PERFORM get_variant_values TABLES r_vbeln
                                    r_auart
                                    r_fkart
                                    r_vbtyp
                                    r_vkorg
                                    r_vtweg
                                    r_spart
                                    r_kunag
                                    r_kdgrp
                                    r_audat
                                    r_erdat
                                    r_aedat
                                    r_vdatu
                                    r_matnr
                                    r_svbtyp
                             USING  rl_variant-low
                                   'ZPR_RFC_INVOICE_HIST_VARIANT'.
*                             CHANGING g_website.

*SOLD TO
  IF sold_to IS NOT INITIAL.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = sold_to
      IMPORTING
        output = gs_kunnr-low.
    gs_kunnr-sign = 'I'.
    gs_kunnr-option = 'EQ'.
    APPEND gs_kunnr TO gr_kunnr.
  ELSE.
    gr_kunnr[] = r_kunnr[].
  ENDIF.

*Ship To
  IF ship_to IS NOT INITIAL.
    rl_shipto-sign = 'I'.
    rl_shipto-option = 'EQ'.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = ship_to
      IMPORTING
        output = rl_shipto-low.
    APPEND rl_shipto TO r_shipto.
  ENDIF.

*DATE
  CLEAR gs_erdat.
  READ TABLE r_erdat INTO gs_erdat INDEX 1.
  IF from_date IS NOT INITIAL.
    rl_shipdt-high = to_ship_date.
    gs_erdat-low = from_date.
  ENDIF.
  IF to_date IS INITIAL.
    gs_erdat-high = sy-datum.
  ELSE.
    gs_erdat-high = to_date.
  ENDIF.
  gs_erdat-sign = 'I'.
  gs_erdat-option = 'BT'.
  REFRESH gr_erdat.
  APPEND gs_erdat TO gr_erdat.

*Ship Date
  IF from_ship_date IS NOT INITIAL OR to_ship_date IS NOT INITIAL.
    rl_shipdt-sign = 'I'.
    rl_shipdt-option = 'BT'.
    rl_shipdt-low = from_ship_date.
    IF to_ship_date IS INITIAL.
      rl_shipdt-high = '99999999'.
    ELSE.
      rl_shipdt-high = to_ship_date.
    ENDIF.
    APPEND rl_shipdt TO r_shipdt.
  ENDIF.

*INVOICE
  IF from_invoice IS NOT INITIAL.
    PERFORM convert_internal CHANGING from_invoice.
  ENDIF.
  IF to_invoice IS NOT INITIAL.
    PERFORM convert_internal CHANGING to_invoice.
  ENDIF.
  IF from_invoice IS NOT INITIAL OR to_invoice IS NOT INITIAL.
    gs_vbeln-sign = 'I'.
    gs_vbeln-option = 'BT'.
    gs_vbeln-low = from_invoice.
    gs_vbeln-high = to_invoice.
    APPEND gs_vbeln TO gr_vbeln.
  ELSE.
    gr_vbeln[] = r_vbeln[].
  ENDIF.

*SALES ORG
  IF sales_org_in[] IS NOT INITIAL.
    LOOP AT sales_org_in.
      gs_vkorg-sign = 'I'.
      gs_vkorg-option = 'EQ'.
      gs_vkorg-low = sales_org_in-vkorg.
      APPEND gs_vkorg TO gr_vkorg.
    ENDLOOP.
  ELSE.
    gr_vkorg[] = r_vkorg[].
  ENDIF.

*DISTRIBUTION CHANNEL
  IF dist_channel_in[] IS NOT INITIAL.
    LOOP AT dist_channel_in.
      gs_vtweg-sign = 'I'.
      gs_vtweg-option = 'EQ'.
      gs_vtweg-low = dist_channel_in-vtweg.
      APPEND gs_vtweg TO gr_vtweg.
    ENDLOOP.
  ELSE.
    gr_vtweg[] = r_vtweg[].
  ENDIF.

*DIVISION
  IF division_in[] IS NOT INITIAL.
    LOOP AT division_in.
      gs_spart-sign = 'I'.
      gs_spart-option = 'EQ'.
      gs_spart-low = division_in-spart.
      APPEND gs_spart TO gr_spart.
    ENDLOOP.
  ELSE.
    gr_spart[] = r_spart[].
  ENDIF.

*DOCTYPE
  IF doc_type_in[] IS NOT INITIAL.
    LOOP AT doc_type_in.
      gs_fkart-sign = 'I'.
      gs_fkart-option = 'EQ'.
      gs_fkart-low = doc_type_in-fkart.
      APPEND gs_fkart TO gr_fkart.
    ENDLOOP.
  ELSE.
    gr_fkart[] = r_fkart[].
  ENDIF.

*custpo
  IF cust_po IS NOT INITIAL.
    rl_custpo-sign = 'I'.
    rl_custpo-option = 'EQ'.
    rl_custpo-low = cust_po.
    APPEND rl_custpo TO r_custpo.
  ENDIF.

*style
  IF style IS NOT INITIAL.
    rl_style-sign = 'I'.
    rl_style-option = 'EQ'.
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
      EXPORTING
        input  = style
      IMPORTING
        output = rl_style-low.
    APPEND rl_style TO r_style.
  ENDIF.

*Sales Order
  IF sales_order IS NOT INITIAL.
    rl_order-sign = 'I'.
    rl_order-option = 'EQ'.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = sales_order
      IMPORTING
        output = rl_order-low.
    APPEND rl_order TO r_order.
  ENDIF.

*Web Order Number
  IF from_web_order IS NOT INITIAL OR to_web_order IS NOT INITIAL.
    rl_webord-sign    = 'I'.
    rl_webord-option  = 'BT'.
    rl_webord-low     = from_web_order.
    rl_webord-high    = to_web_order.
    APPEND rl_webord TO r_webord.
  ENDIF.

*Sales Order Date
  IF from_sales_ord_date IS NOT INITIAL OR to_sales_ord_date IS NOT INITIAL.
    rl_sodate-sign = 'I'.
    rl_sodate-option = 'BT'.
    rl_sodate-low = from_sales_ord_date.
    IF to_sales_ord_date IS INITIAL.
      rl_sodate-high = '99999999'.
    ELSE.
      rl_sodate-high = to_sales_ord_date.
    ENDIF.
    APPEND rl_sodate TO r_sodate.
  ENDIF.

          vbeln TYPE vbeln,
        END OF lt_invoices.
  SELECT vbeln INTO TABLE lt_invoices FROM  vrkpa
         WHERE  vkorg  IN gr_vkorg
         AND    fkdat  IN gr_erdat
         AND    vtweg  IN gr_vtweg
         AND    fkart  IN gr_fkart
         AND    kunag  IN gr_kunnr
         AND    vbeln  IN gr_vbeln.

  IF lt_invoices[] IS NOT INITIAL.
    SELECT vbrk~vbeln
           vbrp~posnr
           vbrk~fkart
           vbrk~vbtyp
           vbrk~waerk
           vbrk~vkorg
           vbrk~vtweg
           vbrk~knumv
           vbrk~fkdat
           vbrk~valdt
           vbrk~zterm
           vbrk~netwr
           vbrk~fkdat
           vbrk~kunag
           vbrk~bstnk_vf
           vbrp~vgbel
           vbrp~aubel
           vbrp~matnr
           vbrp~spart
           vbrp~fkimg
           vbak~vbeln
           vbak~auart
           vbak~bsark
           vbak~ihrez
           vbak~bname
           vbak~erdat
           vbak~vdatu
           vbak~gueen
           likp~vbeln
           likp~kunnr
           likp~wadat_ist
           likp~traty
           likp~bolnr
           FROM vbrk AS vbrk
           INNER JOIN vbrp AS vbrp
           ON vbrk~vbeln = vbrp~vbeln
           INNER JOIN vbak AS vbak
           ON vbak~vbeln = vbrp~aubel
           LEFT OUTER JOIN likp AS likp
           ON likp~vbeln = vbrp~vgbel
           INTO TABLE gt_invoice
           FOR ALL ENTRIES IN lt_invoices
     WHERE vbrk~vbeln = lt_invoices-vbeln   AND
           vbrk~fkart IN gr_fkart    AND
           vbrk~vbtyp IN r_vbtyp     AND
           vbrk~vkorg IN gr_vkorg    AND
           vbrk~vtweg IN gr_vtweg    AND
           vbrk~spart IN gr_spart    AND
           vbrk~kunag IN gr_kunnr    AND
           vbrp~matnr IN r_style     AND
           vbrp~aubel IN r_order     AND
           vbrk~fkdat IN gr_erdat    AND
           vbrk~bstnk_vf IN r_custpo AND
*         likp~kunnr    in r_shipto and
*         likp~wadat_ist in r_shipdt and
           vbak~ihrez     IN r_webord AND
           vbak~erdat     IN r_sodate AND
           vbak~auart     IN r_auart  AND
           vbak~vbtyp     IN r_svbtyp.
  ENDIF.

*  READ TABLE gt_invoice WITH KEY auart = 'XDIS' TRANSPORTING NO FIELDS.
*  IF sy-subrc = 0.
*    PERFORM direct_ship_shipto TABLES gt_invoice.
*  ENDIF.

  IF r_shipto[] IS NOT INITIAL OR r_shipdt[] IS NOT INITIAL.
    LOOP AT gt_invoice INTO gs_invoice.
      IF gs_invoice-wadat_ist IN r_shipdt AND gs_invoice-shipto IN r_shipto.
        CONTINUE.
      ELSE.
        DELETE gt_invoice INDEX sy-tabix.
      ENDIF.
    ENDLOOP.
  ENDIF.

  DATA: BEGIN OF ls_hu,
          vpobjkey TYPE vekp-vpobjkey,
        END OF ls_hu.
  DATA: lt_hu LIKE TABLE OF ls_hu.

  LOOP AT gt_invoice INTO gs_invoice WHERE matnr <> 'FREIGHT'.
    ls_totunit-vbeln = gs_invoice-vbeln.
    ls_totunit-fkimg = gs_invoice-fkimg.
    COLLECT ls_totunit INTO lt_totunit.
    ls_hu-vpobjkey = gs_invoice-vgbel.
    COLLECT ls_hu INTO lt_hu.
  ENDLOOP.
  SORT lt_totunit BY vbeln.

  SORT gt_invoice BY vbeln ASCENDING.
  IF gt_invoice[] IS INITIAL.
    EXIT.
  ENDIF.

  IF gt_invoice[] IS NOT INITIAL.

    LOOP AT gt_invoice ASSIGNING FIELD-SYMBOL(<invoice>).
      gs_likp-vbeln = <invoice>-vgbel.
      APPEND gs_likp TO gt_likp.
    ENDLOOP.
    PERFORM get_tracking_numbers TABLES  gt_likp
                                         gt_vekp.

    LOOP AT gt_invoice INTO gs_invoice.
      CLEAR gs_zsxx_pr_invoice.

*  gs_zsxx_pr_invoice-id
*  gs_zsxx_pr_invoice-localeid
*  gs_zsxx_pr_invoice-hostref
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING
          input  = gs_invoice-vbeln
        IMPORTING
          output = gs_zsxx_pr_invoice-invoiceno.
*SEQ
      gs_zsxx_pr_invoice-website            = website.
      gs_zsxx_pr_invoice-salesorg           = gs_invoice-vkorg.
      gs_zsxx_pr_invoice-division           = gs_invoice-spart.
*    perform get_tracking_num using gs_invoice-vgbel
*                             changing gs_zsxx_pr_invoice-trackingno.
*    clear ls_vekp.
*    read table lt_vekp into ls_vekp with key vpobj = '01' vpobjkey = gs_invoice-vgbel binary search.
*    gs_zsxx_pr_invoice-trackingno = ls_vekp-inhalt.
*      gs_zsxx_pr_invoice-trackingno         = gs_invoice-bolnr.
      READ TABLE gt_vekp ASSIGNING FIELD-SYMBOL(<vekp>) WITH KEY vbeln = gs_invoice-delivery BINARY SEARCH. "D02K912974
      IF sy-subrc = 0.                                      "D02K912974
        gs_zsxx_pr_invoice-trackingno = <vekp>-inhalt.      "D02K912974
      ENDIF.                                                "D02K912974
      gs_zsxx_pr_invoice-trxtype            = gs_invoice-vbtyp.
      gs_zsxx_pr_invoice-ordertype          = gs_invoice-auart.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING
          input  = gs_invoice-sale_ord
        IMPORTING
          output = gs_zsxx_pr_invoice-orderno.
      gs_zsxx_pr_invoice-custpotype         = gs_invoice-bsark.
      gs_zsxx_pr_invoice-custpono           = gs_invoice-bstnk_vf.
      gs_zsxx_pr_invoice-webordertype       = gs_invoice-bsark.
      gs_zsxx_pr_invoice-weborderno         = gs_invoice-ihrez.
      gs_zsxx_pr_invoice-dateordered        = gs_invoice-so_date.
      PERFORM format_timestamp USING gs_invoice-vdatu
                                    gc_timefinish
                           CHANGING gs_zsxx_pr_invoice-daterequested.
*  gs_zsxx_pr_invoice-dateshipnotbefor
*  gs_zsxx_pr_invoice-dateshipnotafter
      PERFORM format_timestamp USING gs_invoice-gueen
                                   gc_timefinish
                          CHANGING gs_zsxx_pr_invoice-datecancelafter.
      PERFORM format_timestamp USING gs_invoice-wadat_ist
                                     gc_timestart
                            CHANGING gs_zsxx_pr_invoice-dateshipped.
      PERFORM format_timestamp USING gs_invoice-fkdat
                                     gc_timestart
                            CHANGING gs_zsxx_pr_invoice-dateinvoice.
      gs_zsxx_pr_invoice-dateinvoice = gs_invoice-erdat.

*      IF gs_invoice-valdt NE '00000000'.
*        PERFORM determine_due_dates  USING gs_invoice-valdt
*                                           gs_invoice-zterm
*                                  CHANGING gs_zsxx_pr_invoice-datedue
*                                           gs_zsxx_pr_invoice-dateduediscount
*                                           gs_zsxx_pr_invoice-discountpercent.
*      ELSE.
      PERFORM determine_due_dates  USING gs_invoice-fkdat
                                         gs_invoice-zterm
                                CHANGING gs_zsxx_pr_invoice-datedue
                                         gs_zsxx_pr_invoice-dateduediscount
                                         gs_zsxx_pr_invoice-discountpercent.
*      ENDIF.

      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING
          input  = gs_invoice-kunag
        IMPORTING
          output = gs_zsxx_pr_invoice-soldto.

*gs_zsxx_pr_invoice-soldtoname
*gs_zsxx_pr_invoice-soldtoaddress1
*gs_zsxx_pr_invoice-soldtoaddress2
*gs_zsxx_pr_invoice-soldtoaddress3
*gs_zsxx_pr_invoice-soldtocity
*gs_zsxx_pr_invoice-soldtoregion
*gs_zsxx_pr_invoice-soldtopostalcode
*gs_zsxx_pr_invoice-soldtocountry
      gs_zsxx_pr_invoice-buyer              = gs_invoice-bname.
*gs_zsxx_pr_invoice-email
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING
          input  = gs_invoice-shipto
        IMPORTING
          output = gs_zsxx_pr_invoice-shipto.
*gs_zsxx_pr_invoice-storeno
*gs_zsxx_pr_invoice-shiptoname
*gs_zsxx_pr_invoice-shiptoaddress1
*gs_zsxx_pr_invoice-shiptoaddress2
*gs_zsxx_pr_invoice-shiptoaddress3
*gs_zsxx_pr_invoice-shiptocity
*gs_zsxx_pr_invoice-shiptoregion
*gs_zsxx_pr_invoice-shiptopostalcode
*gs_zsxx_pr_invoice-shiptocountry
*gs_zsxx_pr_invoice-billto
*gs_zsxx_pr_invoice-billtoname
*gs_zsxx_pr_invoice-billtoaddress1
*gs_zsxx_pr_invoice-billtoaddress2
*gs_zsxx_pr_invoice-billtoaddress3
*gs_zsxx_pr_invoice-billtocity
*gs_zsxx_pr_invoice-billtoregion
*gs_zsxx_pr_invoice-billtopostalcode
*gs_zsxx_pr_invoice-billtocountry
      gs_zsxx_pr_invoice-currency           = gs_invoice-waerk.

      gs_zsxx_pr_invoice-termscode          = gs_invoice-zterm.
*    clear ls_t052u.
*    read table lt_t052u into ls_t052u with key zterm = gs_invoice-zterm binary search.
*    gs_zsxx_pr_invoice-termsdescription = ls_t052u.
*gs_zsxx_pr_invoice-shipmethodcode
*gs_zsxx_pr_invoice-shipmethodtext
*Shipmethod
      PERFORM get_shipmethod USING gs_invoice-sale_ord
                                   '000000'
                                   'SP'
                             CHANGING gs_zsxx_pr_invoice-shipmethodcode
                                      gs_zsxx_pr_invoice-shipmethodtext.
*    gs_zsxx_pr_invoice-shipviacode        = gs_invoice-traty.
*    clear ls_tvtyt.
*    read table lt_tvtyt into ls_tvtyt with key traty = gs_likp-traty binary search.

*    gs_zsxx_pr_invoice-freighttermscode   = gs_zsxx_pr_invoice-shipviacode.

*          l_subtotalwholesal type zsxx_pr_invoice-subtotalwholesal,
*          l_discountpercent  type zsxx_pr_invoice-discountpercent,
*          l_subtotalnet      type zsxx_pr_invoice-subtotalnet,
*          l_freight          type zsxx_pr_invoice-freight,
*          l_taxable          type zsxx_pr_invoice-taxable,
*          l_taxes            type zsxx_pr_invoice-taxes,
*          l_rebate           type zsxx_pr_invoice-rebate,
*          l_other            type zsxx_pr_invoice-other.

*    loop at gt_invoice into gs_invoice where vbeln = gs_invoice-vbeln.
*      if gs_vbrp-matnr <> 'FREIGHT'.

      CLEAR ls_totunit.
      READ TABLE lt_totunit INTO ls_totunit WITH KEY vbeln = gs_invoice-vbeln BINARY SEARCH.
      gs_zsxx_pr_invoice-totalunits = ls_totunit-fkimg.

*      l_freight          = l_freight + gs_lineprice-freight.

*  endloop.

      gs_zsxx_pr_invoice-invoicetotalnet = gs_invoice-netwr.
