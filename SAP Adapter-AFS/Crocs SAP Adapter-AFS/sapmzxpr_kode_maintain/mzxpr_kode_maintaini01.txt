MODULE tc_kode_mark INPUT.
  IF tc_kode-line_sel_mode = 1 AND
     g_tc_kode_wa-flag = 'X'.
    LOOP AT g_tc_kode_itab INTO g_tc_kode_wa
      WHERE flag = 'X'.
      g_tc_kode_wa-flag = ''.
      MODIFY g_tc_kode_itab
        FROM g_tc_kode_wa
        TRANSPORTING flag.
    ENDLOOP.
    g_tc_kode_wa-flag = 'X'.
  ENDIF.
  MODIFY g_tc_kode_itab
    FROM g_tc_kode_wa
    INDEX tc_kode-current_line
    TRANSPORTING flag.
ENDMODULE.                    "TC_KODE_mark INPUT

MODULE exit_command INPUT.

  save_okay = ok_code.

  CASE save_okay.
    WHEN 'ZCANCEL'.
      CLEAR:  ok_code.
      LEAVE TO SCREEN 100.
    WHEN 'ZEXIT'.
      CLEAR:  ok_code.
      LEAVE PROGRAM.
    WHEN OTHERS.
*---- do nothing.
  ENDCASE.

ENDMODULE.                 " EXIT_COMMAND  INPUT
