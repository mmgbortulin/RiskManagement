FUNCTION Z_XX_PR_TEXTEDIT.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(TITEL) OPTIONAL
*"     VALUE(I_ROW) TYPE  I DEFAULT 5
*"     VALUE(I_COLUMN) TYPE  I DEFAULT 5
*"  TABLES
*"      TEXTLINES
*"  EXCEPTIONS
*"      USER_CANCELLED
*"----------------------------------------------------------------------

* Initialisieren
  DATA: RESULT   LIKE ITCER,
        LINES    LIKE TLINE OCCURS 0 WITH HEADER LINE.

  MOVE TEXTLINES[] TO MYTABLE[].
  MOVE TEXTLINES[] TO GL_TEXTLINES[].
  MOVE TITEL       TO GL_TITLE.

* Editor screen
*  CALL SCREEN 100 STARTING AT 5 5.
  CALL SCREEN 100 STARTING AT I_COLUMN
                              I_ROW.
  MOVE GL_TEXTLINES[] TO TEXTLINES[].
  FREE: EDITOR,
        GL_TEXTLINES.

ENDFUNCTION.
