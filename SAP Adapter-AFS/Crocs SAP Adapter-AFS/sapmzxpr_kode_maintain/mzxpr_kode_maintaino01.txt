MODULE tc_kode_init OUTPUT.

* set gui status and titlebar
  SET PF-STATUS 'GUI_0100'.
  SET TITLEBAR 'TITLE_0100'.

  CLEAR:  gv_mode,
          sy-ucomm.
  IF g_tc_kode_copied IS INITIAL.
*&spwizard: copy ddic-table 'ZTXX_PR_KODE'
*&spwizard: into internal table 'g_TC_KODE_itab'
    REFRESH: gt_kode,
             g_tc_kode_itab.
*-- technical website
    IF gv_techsite IS INITIAL.
      gv_techsite = '%'.
    ELSE.
      TRANSLATE gv_techsite USING '*%'.
    ENDIF.
*-- Sales Organization
    IF gv_salesorg IS INITIAL.
      gv_salesorg = '%'.
    ELSE.
      TRANSLATE gv_salesorg USING '*%'.
    ENDIF.
*{   INSERT         DV2K907927                                        1
*-- Dist Channel
    IF gv_distchnl IS INITIAL.
      gv_distchnl = '%'.
    ELSE.
      TRANSLATE gv_distchnl USING '*%'.
    ENDIF.
*}   INSERT
*-- Division
    IF gv_division IS INITIAL.
      gv_division = '%'.
    ELSE.
      TRANSLATE gv_division USING '*%'.
    ENDIF.
*-- keyword
    IF gv_keyword IS INITIAL.
      gv_keyword = '%'.
    ELSE.
      TRANSLATE gv_keyword USING '*%'.
    ENDIF.
    SELECT * FROM ztxx_pr_kode
       INTO TABLE gt_kode
       WHERE techsite LIKE gv_techsite    AND
             salesorg LIKE gv_salesorg    AND
             distchnl LIKE gv_distchnl    AND
             division LIKE gv_division    AND
             keyword  LIKE gv_keyword.
    IF sy-subrc = 0.
      SORT gt_kode BY techsite salesorg distchnl division
                      keyword code division
                      keyword code.
      LOOP AT gt_kode ASSIGNING <kode>.
        MOVE-CORRESPONDING <kode> TO g_tc_kode_wa.
        APPEND g_tc_kode_wa TO g_tc_kode_itab.
      ENDLOOP.
    ENDIF.
*-- technical website
    IF gv_techsite CO '% '.
      CLEAR: gv_techsite.
    ELSE.
      TRANSLATE gv_techsite USING '%*'.
    ENDIF.
*-- Sales Organization
    IF gv_salesorg CO '% '.
      CLEAR:  gv_salesorg.
    ELSE.
      TRANSLATE gv_salesorg USING '%*'.
    ENDIF.
*{   INSERT         DV2K907927                                        3
*-- Dist Channel
    IF gv_distchnl IS INITIAL.
      gv_distchnl = '% '.
    ELSE.
      TRANSLATE gv_distchnl USING '%*'.
    ENDIF.
*}   INSERT
*-- Division
    IF gv_division CO '% '.
      CLEAR:  gv_division.
    ELSE.
      TRANSLATE gv_division USING '%*'.
    ENDIF.
*-- keyword
    IF gv_keyword CO '% '.
      CLEAR:  gv_keyword.
    ELSE.
      TRANSLATE gv_keyword USING '%*'.
    ENDIF.
    g_tc_kode_copied = 'X'.
    REFRESH CONTROL 'TC_KODE' FROM SCREEN '0100'.
  ENDIF.
ENDMODULE.                    "TC_KODE_init OUTPUT
