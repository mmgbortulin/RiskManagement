FORM fill_style.
* Fill ZTXX_PR_STYLE HDI table
  CLEAR:  gs_style,
          gt_style[].

  CLEAR:  gv_seq,
          gv_dateinserted.
  CONCATENATE sy-datum
              sy-uzeit
          INTO gv_dateinserted.
  CLEAR: gt_gnr,gt_atnn,gt_cabt,gt_cawn.
  gt_mat[] = gt_mara[].
  SORT gt_mat BY matnr.
  DELETE ADJACENT DUPLICATES FROM gt_mat COMPARING matnr.
  IF NOT gt_mat[] IS INITIAL.
    SELECT * FROM j_3apghd INTO TABLE gt_gnr FOR ALL ENTRIES IN gt_mat
    WHERE j_3apgnr = gt_mat-j_3apgnr.
    IF NOT gt_gnr[] IS INITIAL.
      LOOP AT gt_gnr INTO gs_gnr.
        CLEAR: gs_atnn.
        gs_atnn-j_3abzd = gs_gnr-j_3abzd1.
        gs_atnn-atinn   = gs_gnr-j_3abzd1.
        APPEND gs_atnn TO gt_atnn.
        CLEAR: gs_atnn.
        gs_atnn-j_3abzd = gs_gnr-j_3abzd2.
        gs_atnn-atinn   = gs_gnr-j_3abzd2.
        APPEND gs_atnn TO gt_atnn.
        CLEAR: gs_atnn.
        gs_atnn-j_3abzd = gs_gnr-j_3abzd3.
        gs_atnn-atinn   = gs_gnr-j_3abzd3.
        APPEND gs_atnn TO gt_atnn.
      ENDLOOP.
      DELETE gt_atnn WHERE j_3abzd EQ space.
      IF NOT gt_atnn[] IS INITIAL.
        SELECT * FROM cabnt APPENDING TABLE gt_cabt FOR ALL ENTRIES IN gt_atnn
        WHERE atinn = gt_atnn-atinn
          AND spras = sy-langu.
        SELECT atinn atzhl adzhl atwrt FROM cawn INTO TABLE gt_cawn
        FOR ALL ENTRIES IN gt_atnn WHERE atinn = gt_atnn-atinn.
      ENDIF.
    ENDIF.
  ENDIF.
  LOOP AT gt_mat ASSIGNING <mara>.
    CLEAR: gs_style,gs_cabt,gs_atnn,gs_gnr,gs_gendt,wa_mlist.
*-- set primary key
    gv_seq = gv_seq + 1.
    gs_style-mandt = sy-mandt.
    gs_style-techsite = gs_tablekey-techsite.
    gs_style-vkorg = space.
    gs_style-vtweg = space.
    gs_style-spart = <mara>-spart.
*    gs_style-seq = gv_seq.
*-- set HDI fields
    gs_style-id = gv_seq.
    gs_style-localeid = gc_localeid_en.      "1033: English
*    READ TABLE it_mlist INTO wa_mlist WITH KEY zzmatnr = <mara>-matnr.
*    IF sy-subrc NE 0.
*      WRITE:  <mara>-matnr TO gs_style-style LEFT-JUSTIFIED
*                            USING EDIT MASK '==ALPHA'.
*    ELSE.
*      WRITE:  <mara>-matnr TO gs_style-style LEFT-JUSTIFIED.
*    ENDIF.
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_OUTPUT'
      EXPORTING
        input  = <mara>-matnr
      IMPORTING
        output = gs_style-style.
    gs_style-hostref = gs_style-style.
    gs_style-website = gs_techsite-pr_website.
    PERFORM get_material_description  USING <mara>-matnr
                                            gs_style-localeid
                                   CHANGING gs_style-name.
    gs_style-description = gs_style-name.
    gs_style-zzstyle = gs_style-style.
    gs_style-alpha = <mara>-extwg.
*    GS_STYLE-VENDOR
*    GS_STYLE-SUPPLIERSTYLE
    WRITE: <mara>-ntgew TO gs_style-weight
                        UNIT <mara>-gewei.
*    READ TABLE gt_gnr INTO gs_gnr WITH KEY j_3apgnr = <mara>-j_3apgnr.
*    IF sy-subrc EQ 0.
*      READ TABLE gt_atnn INTO gs_atnn WITH KEY j_3abzd = gs_gnr-j_3abzd2.
*      IF sy-subrc EQ 0.
*        READ TABLE gt_cabt INTO gs_cabt WITH KEY atinn = gs_atnn-atinn.
*        IF sy-subrc EQ 0.
*          gs_style-attributelabel1 = gs_cabt-atbez.
*        ENDIF.
*      ENDIF.
*      READ TABLE gt_atnn INTO gs_atnn WITH KEY j_3abzd = gs_gnr-j_3abzd1.
*      IF sy-subrc EQ 0.
*        READ TABLE gt_cabt INTO gs_cabt WITH KEY atinn = gs_atnn-atinn.
*        IF sy-subrc EQ 0.
*          gs_style-attributelabel2 = gs_cabt-atbez.
*        ENDIF.
*      ENDIF.
*      READ TABLE gt_atnn INTO gs_atnn WITH KEY j_3abzd = gs_gnr-j_3abzd3.
*      IF sy-subrc EQ 0.
*        READ TABLE gt_cabt INTO gs_cabt WITH KEY atinn = gs_atnn-atinn.
*        IF sy-subrc EQ 0.
*          gs_style-attributelabel3 = gs_cabt-atbez.
*        ENDIF.
*      ENDIF.
*      READ TABLE gt_gendt INTO gs_gendt WITH KEY spras    = sy-langu
*                                                 j_3agend = <mara>-j_3agend.
*      IF sy-subrc EQ 0.

*      ENDIF.
*    ENDIF.
    gs_style-attributelabel1 = 'Size'.
    gs_style-attributelabel2 = 'Width'.
    gs_style-attributelabel3 = 'Gender'.
    gs_style-attributelabel4 = 'ColorCode'.
    gs_style-attributelabel5 = 'ColorName'.
*    GS_STYLE-ATTRIBUTELABEL4 = 'Material No.'.
*    GS_STYLE-ATTRIBUTELABEL5
    gs_style-dateinserted = gv_dateinserted.

    APPEND gs_style TO gt_style.
  ENDLOOP.
  MODIFY ztxx_pr_style FROM TABLE gt_style.
  IF sy-subrc = 0.
    COMMIT WORK.
  ELSE.
*---- Records failed to be inserted
    ROLLBACK WORK.
    MESSAGE e100(zplumriver) WITH 'Style'.
  ENDIF.
ENDFORM.                    " FILL_STYLE
