*&---------------------------------------------------------------------*
*&      Form  find_and_replace
*&---------------------------------------------------------------------*
FORM find_and_replace USING    us_find
                               us_replace
                      CHANGING ch_text.
* Description:  find text and replace.  Caution! trailing spaces
*               in find string are ignored.

* Local declarations
  DATA: lv_len     TYPE sy-fdpos,
        lv_findlen TYPE sy-fdpos,
        lv_string  TYPE string,
        lv_string2 TYPE string.

* Begin processing
  SEARCH ch_text FOR us_find.
  IF sy-subrc = 0.
    CLEAR: lv_string,
           lv_string2.
    lv_string = ch_text.
*    describe field us_find length lv_findlen.
    lv_findlen = strlen( us_find ).
    DO.
      SEARCH lv_string FOR us_find.
      IF sy-subrc = 0.
        lv_len = sy-fdpos + lv_findlen.
        CONCATENATE lv_string2
                    lv_string(sy-fdpos)
                    us_replace
               INTO lv_string2.
        SHIFT lv_string LEFT BY lv_len PLACES.
      ELSE.
        EXIT.
      ENDIF.
    ENDDO.
*-- nothing left to replace...build rest of the string
    CONCATENATE lv_string2
                lv_string
           INTO lv_string2.
    ch_text = lv_string2.
  ENDIF.

ENDFORM.                    " find_and_replace

FORM sub_get_missed_rec CHANGING ch_rc TYPE sy-subrc.

  TYPES:  BEGIN OF t_vbrk,
            vbeln TYPE vbeln,
          END OF t_vbrk,

          BEGIN OF t_vbeln,
            sign   TYPE char1,
            option TYPE char2,
            low    TYPE vbeln,
            high   TYPE vbeln,
          END OF t_vbeln,

          BEGIN OF t_inv,
            zzinvoice TYPE vbeln,
          END OF t_inv,

          BEGIN OF t_date,
            sign   TYPE char1,
            option TYPE char2,
            low    TYPE datum,
            high   TYPE datum,
          END OF t_date,

          BEGIN OF t_date_inst,
            sign   TYPE char1,
            option TYPE char2,
            low    TYPE zplumriver_dateinserted,
            high   TYPE zplumriver_dateinserted,
          END OF t_date_inst.

  DATA: it_vbrk      TYPE TABLE OF t_vbrk,
        it_date      TYPE TABLE OF t_date,
        it_inv       TYPE TABLE OF t_inv,
        it_vbeln     TYPE TABLE OF t_vbeln,
        it_date_inst TYPE TABLE OF t_date_inst.

  DATA: wa_date_inst TYPE t_date_inst,
        wa_date      TYPE t_date.

  DATA:  l_dat   TYPE sy-datum,
         l_mdate TYPE zplumriver_dateordered,
         l_mxdat TYPE zplumriver_dateordered.

  FIELD-SYMBOLS: <fs_vbrk>  TYPE t_vbrk,
                 <fs_vbeln> TYPE t_vbeln,
                 <fs_inv>   TYPE t_inv.

  IF p_nomon IS NOT INITIAL.
    CALL FUNCTION 'OIUPR_DATE_SUBTRACT_MONTH'
      EXPORTING
        date       = sy-datum
        months     = p_nomon
      IMPORTING
        date_e     = l_dat
      EXCEPTIONS
        date_error = 1
        OTHERS     = 2.
    CONCATENATE l_dat gc_timestart INTO l_mdate.
    CONCATENATE sy-datum gc_timefinish INTO l_mxdat.
    CLEAR: wa_date_inst,it_date_inst,wa_date,it_date.
    wa_date-sign = wa_date_inst-sign = c_i.
    wa_date-option = wa_date_inst-option = c_bt.
    wa_date_inst-low = l_mdate.
    wa_date_inst-high = l_mxdat.
    wa_date-low = l_dat.
    wa_date-high = sy-datum.
    APPEND wa_date_inst TO it_date_inst.
    APPEND wa_date TO it_date.
    IF NOT s_erdat[] IS INITIAL.
      APPEND LINES OF s_erdat[] TO it_date[].
    ENDIF.
    SELECT zzinvoice FROM ztxx_pr_invoice INTO TABLE it_inv
    WHERE techsite = p_tsite
        AND vkorg    IN s_vkorg
        AND vtweg    IN s_vtweg
        AND spart    IN s_spart
        AND dateinserted IN it_date_inst[].
    LOOP AT it_inv ASSIGNING <fs_inv>.
      APPEND INITIAL LINE TO it_vbeln ASSIGNING <fs_vbeln>.
      <fs_vbeln>-sign = c_i.
      <fs_vbeln>-option = c_eq.
      <fs_vbeln>-low = <fs_inv>-zzinvoice.
    ENDLOOP.
    SELECT vbeln FROM vbrk INTO TABLE it_vbrk
    WHERE vbeln IN s_vbeln    AND
                 fkart IN s_fkart    AND
                 vbtyp IN s_vbtyp    AND
                 vkorg IN s_vkorg    AND
                 vtweg IN s_vtweg    AND
                 kdgrp IN s_kdgrp    AND
                 spart IN s_spart    AND
                 kunag IN s_kunag   AND
                 fkdat IN s_fkdat    AND
                 erdat IN it_date[].
    IF sy-subrc EQ 0.
      SORT it_vbrk BY vbeln.
      IF it_vbeln[] IS NOT INITIAL.
        DELETE it_vbrk WHERE vbeln IN it_vbeln[].
      ENDIF.
    ENDIF.
  ELSE.
    IF NOT s_erdat[] IS INITIAL.
      APPEND LINES OF s_erdat[] TO it_date[].
    ENDIF.
    SELECT zzinvoice FROM ztxx_pr_invoice INTO TABLE it_inv
    WHERE techsite = p_tsite
        AND vkorg    IN s_vkorg
        AND vtweg    IN s_vtweg
        AND spart    IN s_spart.
    LOOP AT it_inv ASSIGNING <fs_inv>.
      APPEND INITIAL LINE TO it_vbeln ASSIGNING <fs_vbeln>.
      <fs_vbeln>-sign = c_i.
      <fs_vbeln>-option = c_eq.
      <fs_vbeln>-low = <fs_inv>-zzinvoice.
    ENDLOOP.
    SELECT vbeln FROM vbrk INTO TABLE it_vbrk
    WHERE vbeln IN s_vbeln    AND
                 fkart IN s_fkart    AND
                 vbtyp IN s_vbtyp    AND
                 vkorg IN s_vkorg    AND
                 vtweg IN s_vtweg    AND
                 kdgrp IN s_kdgrp    AND
                 spart IN s_spart    AND
                 kunag IN s_kunag   AND
                 fkdat IN s_fkdat    AND
                 erdat IN it_date[].
    IF sy-subrc EQ 0.
      SORT it_vbrk BY vbeln.
      IF it_vbeln[] IS NOT INITIAL.
        DELETE it_vbrk WHERE vbeln IN it_vbeln[].
      ENDIF.
    ENDIF.
  ENDIF.
  IF it_vbrk[] IS NOT INITIAL.
    SELECT *
    FROM vbrk INTO TABLE gt_vbrk FOR ALL ENTRIES IN it_vbrk
    WHERE vbeln EQ it_vbrk-vbeln  AND
          fkart IN s_fkart    AND
          vbtyp IN s_vbtyp    AND
          vkorg IN s_vkorg    AND
          vtweg IN s_vtweg    AND
          kdgrp IN s_kdgrp    AND
          spart IN s_spart    AND
          kunag IN s_kunag   AND
          fkdat IN s_fkdat.
* set return code for subsequent processing
    ch_rc = sy-subrc.
  ELSE.
    ch_rc = 4.
  ENDIF.

ENDFORM.                    " SUB_GET_MISSED_REC
