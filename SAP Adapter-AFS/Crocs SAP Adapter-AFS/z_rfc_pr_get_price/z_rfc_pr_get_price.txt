FUNCTION z_rfc_pr_get_price.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IM_SOLDTO) TYPE  KUNNR
*"     VALUE(IM_VKORG) TYPE  VKORG OPTIONAL
*"     VALUE(IM_VTWEG) TYPE  VTWEG OPTIONAL
*"     VALUE(IM_SPART) TYPE  SPART OPTIONAL
*"     VALUE(IM_PRICE_DATE) TYPE  DATAB DEFAULT SY-DATUM
*"     VALUE(IM_ORDER_REASON) TYPE  AUGRU OPTIONAL
*"     VALUE(IM_ORDERTYPE) TYPE  AUART OPTIONAL
*"     VALUE(IM_VKALS) TYPE  VKALS DEFAULT 'ZJ3A01'
*"     VALUE(IM_QTY_ORDERED) TYPE  KWMENG OPTIONAL
*"  TABLES
*"      T_MATERIALS STRUCTURE  ZPR_STR_MATNR
*"      COND_TYPE STRUCTURE  ZSPR_COND OPTIONAL
*"      T_PRICES STRUCTURE  ZPR_STR_PRICES OPTIONAL
*"----------------------------------------------------------------------
*=======================================================================
* ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
*  PlumRiver LLC
*  Delivered:  September, 2012
* ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
*=======================================================================

  TYPES:
    BEGIN OF type_knvv,
      kunnr TYPE kunnr,
      vkorg TYPE vkorg,
      vtweg TYPE vtweg,
      spart TYPE spart,
      kalks TYPE kalks,
      pltyp TYPE pltyp,
      waers TYPE waers,
      land1 TYPE land1,
    END OF type_knvv.

  DATA: it_knvv      TYPE TABLE OF type_knvv,
        wa_knvv      TYPE type_knvv,
        pi_i_spr     TYPE pispr,
        wa_materials TYPE zpr_str_matnr,                    "#EC ENHOK
        it_pe_komv   TYPE TABLE OF komv,
        wa_pe_komv   TYPE komv,
        wa_prices    TYPE zpr_str_prices.                   "#EC ENHOK

  CLEAR: it_vkorg,it_vtweg,it_spart,wa_vkorg,wa_vtweg,wa_spart,wa_tvak,wa_t683v,wa_komk,wa_komp.
  IF NOT im_order_reason IS INITIAL.
    v_augru = im_order_reason.
  ENDIF.
  IF NOT im_vkorg IS INITIAL.
    wa_vkorg-sign   = c_i.
    wa_vkorg-option = c_eq.
    wa_vkorg-low    = im_vkorg.
    APPEND wa_vkorg TO it_vkorg.
  ENDIF.
  IF NOT im_vtweg IS INITIAL.
    wa_vtweg-sign   = c_i.
    wa_vtweg-option = c_eq.
    wa_vtweg-low    = im_vtweg.
    APPEND wa_vtweg TO it_vtweg.
  ENDIF.
  IF NOT im_spart IS INITIAL.
    wa_spart-sign   = c_i.
    wa_spart-option = c_eq.
    wa_spart-low    = im_spart.
    APPEND wa_spart TO it_spart.
  ENDIF.
  IF NOT im_ordertype IS INITIAL.
    SELECT SINGLE auart kalvg FROM tvak INTO wa_tvak WHERE auart = im_ordertype.
  ENDIF.
* Get customer related information
  IF im_soldto IS NOT INITIAL.

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = im_soldto
      IMPORTING
        output = im_soldto.

    SELECT a~kunnr a~vkorg a~vtweg a~spart a~kalks a~pltyp a~waers b~land1
      INTO TABLE it_knvv
      FROM knvv AS a INNER JOIN kna1 AS b
                ON a~kunnr = b~kunnr
     WHERE a~kunnr =  im_soldto
       AND a~vkorg IN it_vkorg
       AND a~vtweg IN it_vtweg
       AND a~spart IN it_spart.

    READ TABLE it_knvv INTO wa_knvv INDEX 1.

    SELECT SINGLE * FROM t683v INTO wa_t683v WHERE vkorg IN it_vkorg
                                               AND vtweg IN it_vtweg
                                               AND spart IN it_spart
                                               AND kalvg EQ wa_tvak-kalvg
                                               AND kalks EQ wa_knvv-kalks.
  ENDIF.

  CHECK it_knvv IS NOT INITIAL.

* there could be multiple sales org for customer, so for every material
* sales org combinition get price.
  LOOP AT t_materials INTO wa_materials.

    CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
      EXPORTING
        input        = wa_materials-material
      IMPORTING
        output       = wa_materials-material
      EXCEPTIONS
        length_error = 1
        OTHERS       = 2.
    CALL FUNCTION 'CONVERSION_EXIT_EAN11_INPUT'
      EXPORTING
        input  = wa_materials-upc
      IMPORTING
        output = wa_materials-upc.
    SELECT SINGLE extwg FROM mara INTO v_extwg WHERE matnr = wa_materials-material.
    IF NOT wa_materials-upc IS INITIAL.
      SELECT SINGLE j_3akordx FROM mean INTO v_grid_size WHERE ean11 = wa_materials-upc.
    ENDIF.
    LOOP AT it_knvv INTO wa_knvv.

      CLEAR: pi_i_spr, it_pe_komv[], wa_pe_komv.

      pi_i_spr-matnr = wa_materials-material.  "Material
      pi_i_spr-ean11 = wa_materials-upc.       "UPC Code
      pi_i_spr-vkorg = wa_knvv-vkorg.          "Sales ORg
      pi_i_spr-vtweg = wa_knvv-vtweg.          "Dist ch
      pi_i_spr-spart = wa_knvv-spart.          "Division
      pi_i_spr-kunnr = im_soldto.              "sold to
      pi_i_spr-prsdt = im_price_date.          "Price date
      pi_i_spr-waerk = wa_knvv-waers.
      pi_i_spr-kwaer = wa_knvv-waers.
      IF NOT wa_t683v-kalsm IS INITIAL.
        pi_i_spr-vkals = wa_t683v-kalsm.         "Configured Pricing Procedure
      ELSE.
        pi_i_spr-vkals = im_vkals.               "Pricing Procedure
      ENDIF.
      pi_i_spr-mgame = im_qty_ordered.         "Quantity Ordered
      pi_i_spr-pltyp = wa_knvv-pltyp.          "Price List Type
      pi_i_spr-tabkz = 'X'.
      pi_i_spr-vorga = '50'.

      CALL FUNCTION 'Z_PR_GET_PRICE'
        EXPORTING
          pi_i_spr                       = pi_i_spr
*         PI_USE_BUFFER                  = 'X'
          auart_sd                       = im_ordertype
          land1                          = wa_knvv-land1
        IMPORTING
*         PE_I_SPR                       =
          pe_i_komk                      = wa_komk
          pe_i_komp                      = wa_komp
        TABLES
          pe_t_komv                      = it_pe_komv
*         PE_T_LIFE                      =
*         PE_T_SCALE                     =
        EXCEPTIONS
          material_not_found             = 1
          no_authority                   = 2
          no_bukrs_found                 = 3
          no_sales_price_found           = 4
          org_structure_not_completed    = 5
          plant_not_found                = 6
          vkorg_not_found                = 7
          calculation_scheme_not_correct = 8
          pricing_error                  = 9
          other_error                    = 10
          OTHERS                         = 11.
      IF sy-subrc <> 0.
*       MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
*          WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      ELSE.
*        READ TABLE it_pe_komv INTO wa_pe_komv WITH KEY kschl = 'J3AP'.
        CLEAR wa_prices.
        wa_prices-soldto = im_soldto.               "Sold-to customer
        wa_prices-country = wa_knvv-land1.          "sold to country
        wa_prices-salesorg = wa_knvv-vkorg.         "sales org
*         wa_prices-companycode                       "company code
        wa_prices-material = wa_materials-material. "material
        wa_prices-upc = wa_materials-upc.           "UPC
        wa_prices-ordertype = im_ordertype.         "Order type
*        wa_prices-conditiontype = wa_pe_komv-kschl. "Condition type
*        PERFORM convert_price USING wa_komp-kzwi5
        PERFORM convert_price USING wa_komp-kzwi1
                                    wa_komk-waerk
                              CHANGING wa_prices-pricewholesale.
*        wa_prices-pricewholesale = wa_komp-kzwi5."gross price
*        wa_prices-pricenet = wa_komp-netwr.      "net price
        PERFORM convert_price USING wa_komp-netwr
                                    wa_komk-waerk
                              CHANGING wa_prices-pricenet.
        wa_prices-currency = wa_komk-waerk.      "currency
        wa_prices-unit = wa_komp-meins.          "sales unit
        READ TABLE it_pe_komv INTO wa_pe_komv WITH KEY kschl = 'MSRP'.
        IF sy-subrc EQ 0.
*          wa_prices-pricemsrp      = wa_pe_komv-kbetr. "MSRP price.
          PERFORM convert_price USING wa_pe_komv-kbetr
                                    wa_komk-waerk
                              CHANGING wa_prices-pricemsrp.
        ENDIF.
        APPEND wa_prices TO t_prices.
      ENDIF.
      CLEAR wa_materials.
    ENDLOOP.
  ENDLOOP.
ENDFUNCTION.
