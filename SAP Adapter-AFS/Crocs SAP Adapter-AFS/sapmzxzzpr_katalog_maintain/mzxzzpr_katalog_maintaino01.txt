    refresh: gt_katalog,
             g_TC_KATALOG_itab.
    if gv_techsite is initial.

    else.
      translate gv_techsite using '*%'.
    endif.
    select * from ZTXX_PR_KATALOG
       into corresponding fields
       of table gt_katalog
       where techsite like gv_techsite.
*       of table g_TC_KATALOG_itab.
    if sy-subrc = 0.
      sort gt_Katalog by techsite catalogx localeid.
      loop at gt_katalog assigning <katalog>.
        move-corresponding <katalog> to g_TC_KATALOG_wa.
        append g_TC_KATALOG_wa to g_TC_KATALOG_itab.
      endloop.
    endif.
    if GV_TECHSITE co '% '.
      clear:  gv_techsite.
    else.
