*&spwizard: input module for tc 'TC_KATALOG'. do not change this line!
*&spwizard: mark table
module TC_KATALOG_mark input.
  if TC_KATALOG-line_sel_mode = 1 and
     g_TC_KATALOG_wa-flag = 'X'.
     loop at g_TC_KATALOG_itab into g_TC_KATALOG_wa
       where flag = 'X'.
       g_TC_KATALOG_wa-flag = ''.
       modify g_TC_KATALOG_itab
         from g_TC_KATALOG_wa
         transporting flag.
     endloop.
     g_TC_KATALOG_wa-flag = 'X'.
  endif.
  modify g_TC_KATALOG_itab
    from g_TC_KATALOG_wa
    index TC_KATALOG-current_line
    transporting flag.
endmodule.

MODULE VALIDATE_KEYFIELDS INPUT.
* VALIDATE THE PRIMARY KEY

  IF GV_MODE+1(1) <> 'D'.   "key is not already locked from change
    SELECT SINGLE MANDT
             from ZTXX_PR_KATALOG into GS_KATALOG
             where techsite = GS_KATALOG-techsite     and
                   catalogx = GS_KATALOG-catalogx     and
                   localeid = GS_KATALOG-localeid.
    if sy-subrc = 0.
*---- Catalog already exists.  Make them create a different one
      message E202(ZPLUMRIVER) with GS_KATALOG-techsite
                                    GS_KATALOG-catalogx
                                    GS_KATALOG-localeid.
    else.
*---- New catalog is being created... lock it down.
      GV_MODE+1(1) = 'D'.
    endif.   "sy-subrc = 0   [ON  SELECT SINGLE MANDT]
  ENDIF.   "GV_MODE+1(1) <> 'D'

ENDMODULE.                 " VALIDATE_KEYFIELDS  INPUT
