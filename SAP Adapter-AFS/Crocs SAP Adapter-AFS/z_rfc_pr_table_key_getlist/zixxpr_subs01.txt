*&---------------------------------------------------------------------*
*&      Form  GET_TECHSITES
*&---------------------------------------------------------------------*
FORM GET_TECHSITES TABLES   TA_TECHSITE STRUCTURE GS_TECHSITE
                            TA_RETURN STRUCTURE GS_RET1
                   USING    US_WEBSITE
                CHANGING    CH_RC.

* LOCAL VARIABLES
  DATA:  LV_MSGID TYPE SYMSGID,
         LV_MSGV1 TYPE SYMSGV.

* Refresh tables
  refresh:  TA_TECHSITE,
            TA_RETURN.

* SET INITIAL VALUE FOR RETURN CODE
  CH_RC = 99.

* GET list of SAP technical sites.
  SELECT  WEBSITE PR_WEBSITE URL
      FROM ZTXX_PR_WEBSITE
      INTO TABLE TA_TECHSITE
      WHERE PR_WEBSITE LIKE US_WEBSITE.
  IF SY-SUBRC = 0.
    CLEAR: CH_RC.
  ELSE.
    CH_RC = SY-SUBRC.
*-- RETURN MESSAGE THAT THE TECHNICAL SITE COULD NOT BE DETERMINED.
    CLEAR:  GS_RET1.
    LV_MSGID = 'ZPLUMRIVER'.
    LV_MSGV1 = US_WEBSITE.
    PERFORM SET_RETURN_MESSAGE  USING  'E'
                                       LV_MSGID
                                       '010'
                                       LV_MSGV1
                                       space
                                       space
                                       space
                             CHANGING  GS_RET1.
    APPEND GS_RET1 TO TA_RETURN.
  ENDIF.

ENDFORM.                    " GET_TECHSITES


*&---------------------------------------------------------------------*
*&      Form  SET_SELECTION_DEFAULTS
*&---------------------------------------------------------------------*
FORM SET_SELECTION_DEFAULTS USING    US_VKORG  type vkorg
                                     US_VTWEG  type VTWEG
                                     US_SPART  type SPART
                         CHANGING   CH_VKORG  type vkorg
                                    CH_VTWEG  type VTWEG
                                    CH_SPART  type SPART.
* Description:  Set default values for the several import parameters

*-- Sales Organization
  If us_vkorg = '*'.
    ch_vkorg = '%'.
  else.
    ch_vkorg = us_vkorg.
  endif.
*-- Distribution channel
  If us_vtweg = '*'.
    ch_vtweg = '%'.
  else.
    ch_vtweg = us_vtweg.
  endif.
*-- Division
  If us_spart = '*'.
    ch_spart = '%'.
  else.
    ch_spart = us_spart.
  endif.

ENDFORM.                    " SET_SELECTION_DEFAULTS


*&---------------------------------------------------------------------*
*&      Form  SET_RETURN_MESSAGE
*&---------------------------------------------------------------------*
FORM SET_RETURN_MESSAGE USING    us_mgtyp  type  BAPI_MTYPE
                                 us_MSGID  type  symsgid
                                 us_msgno  type  symsgno
                                 us_MSGV1  type  symsgv
                                 us_MSGV2  type  symsgv
                                 us_MSGV3  type  symsgv
                                 us_MSGV4  type  symsgv
                        CHANGING CH_RET1 type bapiret1.
* Creat Return message using BAPIRET1 structure

  CALL FUNCTION 'BALW_BAPIRETURN_GET1'
    EXPORTING
      TYPE             = us_mgtyp
      CL               = us_msgid
      NUMBER           = us_msgno
      PAR1             = us_msgv1
      PAR2             = us_msgv2
      PAR3             = us_msgv3
      PAR4             = us_msgv4
    IMPORTING
      BAPIRETURN       =  GS_RET1.

ENDFORM.                    " SET_RETURN_MESSAGE
