*&---------------------------------------------------------------------*
*&      Form  find_and_replace
*&---------------------------------------------------------------------*
FORM find_and_replace USING    us_find
                               us_replace
                      CHANGING ch_text.
* Description:  find text and replace.  Caution! trailing spaces
*               in find string are ignored.

* Local declarations
  data: lv_len type sy-fdpos,
        lv_findlen type sy-fdpos,
        lv_string type string,
        lv_string2 type string.

* Begin processing
  search ch_text for us_find.
  if sy-subrc = 0.
    clear: lv_string,
           lv_string2.
    lv_string = ch_text.
*    describe field us_find length lv_findlen.
    lv_findlen = strlen( us_find ).
    do.
      search lv_string for us_find.
      if sy-subrc = 0.
        lv_len = sy-fdpos + lv_findlen.
        concatenate lv_string2
                    lv_string(sy-fdpos)
                    us_replace
               into lv_string2.
        shift lv_string left by lv_len places.
      else.
        exit.
      endif.
    enddo.
*-- nothing left to replace...build rest of the string
    concatenate lv_string2
                lv_string
           into lv_string2.
    ch_text = lv_string2.
  endif.

ENDFORM.                    " find_and_replace


*&---------------------------------------------------------------------*
*&      Form  transfer_file_to_desktop
*&---------------------------------------------------------------------*
FORM transfer_file_to_desktop USING    us_source type string
                                       us_target type string.
* Description:  transfer source file to target location (desktop).
*               The source file name does not have to match target
*               file name.

* local declarations
  data:  lv_source type SAPB-SAPPFAD,
         lv_target type SAPB-SAPPFAD.

* begin processing
  lv_source = us_source.
  lv_target = us_target.
  CALL FUNCTION 'ARCHIVFILE_SERVER_TO_CLIENT'
    EXPORTING
      path             =  lv_source
      TARGETPATH       =  lv_target
    EXCEPTIONS
      ERROR_FILE       = 1
      OTHERS           = 2.

  IF sy-subrc <> 0 and sy-batch = space.
    MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.


ENDFORM.                    " transfer_file_to_desktop


*&---------------------------------------------------------------------*
*&      Form  transfer_file_to_appserver
*&---------------------------------------------------------------------*
FORM transfer_file_to_appserver USING    us_source type string
                                         us_target type string.
* Description:  transfer source file to application server location.
*               The source file name does not have to match target
*               file name.

* local declarations
  data:  lv_source type SAPB-SAPPFAD,
         lv_target type SAPB-SAPPFAD.

* begin processing
  lv_source = us_source.
  lv_target = us_target.
  CALL FUNCTION 'ARCHIVFILE_SERVER_TO_SERVER'
    EXPORTING
      SOURCEPATH       =  lv_source
      TARGETPATH       =  lv_target
    EXCEPTIONS
      ERROR_FILE       = 1
      OTHERS           = 2.

  IF SY-SUBRC <> 0.
    MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.

ENDFORM.                    " transfer_file_to_appserver
