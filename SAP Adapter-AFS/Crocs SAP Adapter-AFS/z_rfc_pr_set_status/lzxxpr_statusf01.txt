*&---------------------------------------------------------------------*
*&      Form  SET_NEW_STATUS
*&---------------------------------------------------------------------*
FORM SET_NEW_STATUS TABLES  UT_MESSAGE structure BAPIRET1
                    USING    USS_STATUS  type ZTXX_PR_HSTATUS
                             USS_TECHSITE like GS_TECHSITE
                             US_Target type ZPLUMRIVER_PSTATUS
                 CHANGING    CH_NEWSTATUS type ZPLUMRIVER_PSTATUS.
* DESCRIPTION:  Set new processing status.

*// Local Declaration
  DATA: LS_LOG type ZTXX_PR_LSTATUS.

*// Begin Processing
* set message class globally (for use within this subroutine)
  gv_msgid = 'ZPLUMRIVER'.

* BUILD LOG ENTRY (IT DOESN'T MATTER WHETHER ZTXX_PR_HSTATUS RECORD
* WAS FOUND IN PREVIOUS SELECT SINGLE STEP OR NOT).
  LS_LOG-MANDT = SY-MANDT.
  LS_LOG-TECHSITE = USS_TECHSITE-WEBSITE.
  LS_LOG-PROCESS_DATE = SY-DATUM.
  LS_LOG-PROCESS_TIME = SY-UZEIT.
  LS_LOG-NEW_STATUS = US_Target.
  LS_LOG-OLD_STATUS = USS_STATUS-STATUS.
* until update (tentatively) assign CH_NEWSTATUS as the "old" status.
  CH_NEWSTATUS = LS_LOG-OLD_STATUS.
* SET NEW STATUS
  IF LS_LOG-NEW_STATUS = LS_LOG-OLD_STATUS.
*-- NO STATUS CHANGE... EXIT WITHOUT UPDATE.
    clear:  gs_ret1.
    gv_msgv1 = US_TARGET.
    perform SET_RETURN_MESSAGE using 'S' gv_msgid '250'
                                     gv_msgv1 space space space
                          changing   gs_ret1.
    append gs_ret1 to ut_message.
  ELSE.
*-- UPDATE WITH NEW STATUS
    USS_STATUS-MANDT = LS_LOG-MANDT.
    USS_STATUS-TECHSITE = LS_LOG-TECHSITE.
    USS_STATUS-STATUS = LS_LOG-NEW_STATUS.
    CONCATENATE LS_LOG-PROCESS_DATE
                LS_LOG-PROCESS_TIME
             INTO USS_STATUS-DATEINSERTED.
    IF NOT USS_STATUS IS INITIAL.
      MODIFY ZTXX_PR_HSTATUS FROM USS_STATUS.
      IF SY-SUBRC = 0.
*------ HEADER STATUS MODIFIED (PENDING COMMIT)
        INSERT ZTXX_PR_LSTATUS FROM LS_LOG.
        IF SY-SUBRC = 0.
          COMMIT WORK.
          CH_NEWSTATUS = LS_LOG-NEW_STATUS.    "set to new status
          clear:  gs_ret1.
          gv_msgv1 = LS_LOG-OLD_STATUS.
          gv_msgv2 = USS_STATUS-STATUS.
          perform SET_RETURN_MESSAGE using 'S' gv_msgid '253'
                                           gv_msgv1 gv_msgv2
                                           space space
                                changing   gs_ret1.
          append gs_ret1 to ut_message.
        ELSE.
          ROLLBACK WORK.
          clear:  gs_ret1.
          gv_msgv1 = 'ZTXX_PR_LSTATUS'.
          perform SET_RETURN_MESSAGE using 'S' gv_msgid '254'
                                           gv_msgv1 space space space
                                changing   gs_ret1.
          append gs_ret1 to ut_message.
        ENDIF.   "SY-SUBRC = 0  [ON INSERT ZTXX_PR_LSTATUS...]
      ELSE.
*------ STATUS NOT MODIFIED
        clear:  gs_ret1.
        gv_msgv1 = 'ZTXX_PR_HSTATUS'.
        perform SET_RETURN_MESSAGE using 'S' gv_msgid '254'
                                         gv_msgv1 space space space
                              changing   gs_ret1.
        append gs_ret1 to ut_message.
      ENDIF.
    ENDIF.   "NOT USS_STATUS IS INITIAL
  ENDIF.   "LS_LOG-NEW_STATUS = LS_LOG-OLD_STATUS

ENDFORM.                    " SET_NEW_STATUS
