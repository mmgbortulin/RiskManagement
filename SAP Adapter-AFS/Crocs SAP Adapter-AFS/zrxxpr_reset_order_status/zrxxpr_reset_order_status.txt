TYPES: BEGIN OF typ_orderlin,
         techsite TYPE ztxx_pr_orderlin-techsite,
         vkorg    TYPE vkorg,

         spart    TYPE spart,
         zzorder  TYPE vbeln,
         status   TYPE ztxx_pr_orderlin-status,
       END OF typ_orderlin.

DATA: gt_orderlin         TYPE TABLE OF typ_orderlin,
      gs_orderlin         TYPE typ_orderlin,
      gt_orderhdr         TYPE TABLE OF ztxx_pr_orderhdr,
      gs_orderhdr         TYPE ztxx_pr_orderhdr,
      gt_ztxx_pr_orderhdr TYPE TABLE OF ztxx_pr_orderhdr,

      gt_ztxx_pr_orderlin TYPE TABLE OF ztxx_pr_orderlin,
      gs_ztxx_pr_orderlin TYPE ztxx_pr_orderlin.

START-OF-SELECTION.

    SELECT * FROM  ztxx_pr_orderhdr INTO TABLE gt_orderhdr
           WHERE    techsite      in s_tsite
             AND    vkorg         in s_vkorg
             AND    vtweg         in s_vtweg
             AND    spart         in s_spart
             AND    zzorder       in s_vbeln
             AND    soldto        IN s_kunag
             AND    orderstatus   IN s_stat
             AND    dateinserted  IN s_date.

  SELECT  techsite
          vkorg
          vtweg
          spart
          zzorder
          status  FROM  ztxx_pr_orderlin INTO TABLE gt_orderlin
     WHERE  techsite        IN s_tsite
       AND    vkorg         IN s_vkorg
       AND    vtweg         IN s_vtweg
       AND    spart         IN s_spart
       AND    zzorder       IN s_vbeln
       AND    dateinserted  IN s_date.

  MOVE-CORRESPONDING gt_orderlin[] TO gt_ztxx_pr_orderlin[].
  FREE: gt_orderlin.
  MOVE gt_orderhdr[] TO gt_ztxx_pr_orderhdr[].
  FREE: gt_orderhdr.

  IF gt_ztxx_pr_orderhdr[] IS NOT INITIAL.
    CALL FUNCTION 'ZPR_RESET_ORDER_HEADER_STATUS'
      TABLES
        order     = gt_ztxx_pr_orderhdr
        orderline = gt_ztxx_pr_orderlin.

    MODIFY ztxx_pr_orderhdr FROM TABLE gt_ztxx_pr_orderhdr.
    IF sy-subrc = 0.
      WRITE: /1 'Number of Rows modified in ZTXX_PR_ORDERHDR', sy-dbcnt.
      COMMIT WORK.
    ELSE.
      WRITE: /1 'Modify of ZTXX_PR_ORDERLIN Failed.'.
      ROLLBACK WORK.
    ENDIF.
    else.
      write: /1 'No Data Selected'.
  ENDIF.

END-OF-SELECTION.
