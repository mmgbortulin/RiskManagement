FUNCTION CONVERSION_EXIT_ZPRTS_INPUT.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(INPUT)
*"  EXPORTING
*"     VALUE(OUTPUT)
*"----------------------------------------------------------------------
  DATA: LV_STRING TYPE STRING.

  LV_STRING = INPUT.
  IF LV_STRING CO ' -:0123456789'.
*-- convert dashes (-) and colons (:) to blank spaces.
    TRANSLATE LV_STRING USING '- : '.
    CONDENSE LV_STRING NO-GAPS.
    OUTPUT = LV_STRING.
  ELSE.
*-- NO CONVERSION
    OUTPUT = INPUT.
  ENDIF.

ENDFUNCTION.
