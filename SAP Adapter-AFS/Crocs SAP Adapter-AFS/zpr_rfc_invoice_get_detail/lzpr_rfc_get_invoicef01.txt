*----------------------------------------------------------------------*
*      -->US_FKDAT         text

*      <--CH_DUEDATE       text

*      <--CH_PERCENT       text
*----------------------------------------------------------------------*
FORM determine_due_dates USING    us_fkdat
                                  us_zterm
                       CHANGING ch_duedate TYPE zplumriver_datedue
                                ch_discountdate TYPE zplumriver_datedue
                             ch_percent TYPE zplumriver_discountpercent.
* Determine invoice due and discount dates
*// Local Declaration
  DATA:  ls_t052 TYPE t052,
         l_zterm TYPE dzterm.

  DATA:  lv_date    TYPE datum.        "date
*         lv_percent TYPE dzproz.    "discount percentage

  CLEAR: l_zterm.
  SELECT SINGLE zterm FROM  t052 INTO l_zterm WHERE zterm = us_zterm.
  IF sy-subrc EQ 0.
*// Begin Processing
    CALL FUNCTION 'FI_TERMS_OF_PAYMENT_PROPOSE'
      EXPORTING
        i_bldat         = us_fkdat
        i_budat         = us_fkdat
        i_cpudt         = us_fkdat
        i_zfbdt         = us_fkdat
        i_zterm         = us_zterm
      IMPORTING
        e_t052          = ls_t052
      EXCEPTIONS
        terms_not_found = 1
        OTHERS          = 2.

    IF sy-subrc = 0.
      IF ls_t052-zprz1 IS INITIAL.
*---- no discount scenario... SET DUE DATE ONLY
        lv_date = us_fkdat + ls_t052-ztag1.
        PERFORM format_timestamp USING lv_date
                                       gc_timefinish
                              CHANGING ch_duedate.
      ELSE.
*---- check if only one discount date available.
        IF ls_t052-zprz2 IS INITIAL.
*---- only one discount date... SET DATES
*------ Discount Date
          lv_date = us_fkdat + ls_t052-ztag1.
          PERFORM format_timestamp USING lv_date
                                         gc_timefinish
                                CHANGING ch_discountdate.
*------ Discount percentage
          WRITE:  ls_t052-zprz1 TO ch_percent.
*------ Due Date
          lv_date = us_fkdat + ls_t052-ztag2.
          PERFORM format_timestamp USING lv_date
                                         gc_timefinish
                                CHANGING ch_duedate.
        ELSE.
*---- two discount dates available (use 2nd date)... set dates
*------ Discount Date
          lv_date = us_fkdat + ls_t052-ztag2.
          PERFORM format_timestamp USING lv_date
                                         gc_timefinish
                                CHANGING ch_discountdate.
*------ Discount percentage
          WRITE:  ls_t052-zprz2 TO ch_percent.
*------ Due Date
          lv_date = us_fkdat + ls_t052-ztag3.
          PERFORM format_timestamp USING lv_date
                                         gc_timefinish
                                CHANGING ch_duedate.
        ENDIF.
      ENDIF.  "LS_T052-ZPRZ1 is initial
    ELSE.
*   MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
*           WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
    ENDIF.
  ELSE.
  ENDIF.
ENDFORM.                    " DETERMINE_DUE_DATES
*&---------------------------------------------------------------------*
*&      Form  format_timestamp
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->US_DATE       text
*      -->US_TIME       text
*      <--CH_TIMESTAMP  text
*----------------------------------------------------------------------*
FORM format_timestamp USING    us_date TYPE sydatum
                               us_time TYPE syuzeit
                      CHANGING ch_timestamp.
* FORMAT TIMESTAMP FROM DATE (CCYYMMDD) AND TIME (HHMMSS).

  IF us_date IS INITIAL.
*-- NO DATE... SO RETURN BLANK TIMESTAMP VALUE
    CLEAR: ch_timestamp.
  ELSE.
*-- FORMAT DATE/TIME
    CONCATENATE us_date
                us_time
            INTO ch_timestamp.
  ENDIF.

ENDFORM.                    " FORMAT_TIMESTAMP

                            p_item

                     CHANGING p_shipmethod_code
                              p_shipmethod_text.


  CLEAR: p_shipmethod_code,
         p_shipmethod_text.

  SELECT vbpa~lifnr
         lfa1~name1
          INTO ( p_shipmethod_code, p_shipmethod_text )
          FROM vbpa AS vbpa
          INNER JOIN lfa1  AS lfa1 ON lfa1~lifnr = vbpa~lifnr
          WHERE vbpa~vbeln = p_order
            AND vbpa~posnr = p_item
            AND vbpa~parvw = p_partner_type .
  ENDSELECT.

FORM get_tracking_numbers  TABLES   pt_likp STRUCTURE likp
                                    pt_vekp TYPE ttyp_vekp.

  DATA: BEGIN OF ls_hu,
          vpobjkey TYPE vekp-vpobjkey,
        END OF ls_hu.
  DATA: lt_hu LIKE TABLE OF ls_hu.

  CLEAR pt_vekp[].

  LOOP AT pt_likp ASSIGNING FIELD-SYMBOL(<ls_likp>).
    ls_hu-vpobjkey = <ls_likp>-vbeln.
    APPEND ls_hu TO lt_hu.
  ENDLOOP.

  IF lt_hu IS NOT INITIAL.
    SELECT vekp~vpobj,
           vekp~vpobjkey,
           vekp~vhilm_ku,
           vepo~vbeln,
           vepo~posnr,
           vepo~matnr,
           vekp~inhalt,
           vepo~vemng
           FROM vekp AS vekp
           INNER JOIN vepo AS vepo
           ON vekp~venum = vepo~venum
           INTO TABLE @pt_vekp
           FOR ALL ENTRIES IN @lt_hu
           WHERE vekp~vpobjkey = @lt_hu-vpobjkey.
    IF sy-subrc = 0.
      SORT pt_vekp BY vpobjkey vbeln matnr.
      DELETE ADJACENT DUPLICATES FROM pt_vekp COMPARING ALL FIELDS.
    ENDIF.
  ENDIF.
