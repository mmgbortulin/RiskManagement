    ENDDO.
  ENDIF.

    EXPORTING

    IMPORTING
      output = gs_vbrk-vbeln.

  CALL FUNCTION 'RV_INVOICE_DOCUMENT_READ'
    EXPORTING
*     ACTIVITY             = '  '
      konv_read            = 'X'

      vbrk_i               = gs_vbrk
      i_no_authority_check = 'X'

      xkomfk               = gt_xkomfk
      xvbfs                = gt_xvbfs
      xthead               = gt_xthead

* CHANGING

  ENDIF.

              FOR ALL ENTRIES IN gt_xvbrk
              WHERE knumv = gt_xvbrk-knumv
                AND kstat = space        "CB: NOT STATISTICAL
                AND ( kinak = ' ' OR kinak = 'Y' ). "Added by lokesh on 09/13/2016
    SORT gt_konv BY knumv kposn stunr zaehk.

  ENDIF.

                            gt_xvbrp.

    REFRESH gt_likp.
    SELECT * FROM likp INTO TABLE gt_likp
        FOR ALL ENTRIES IN gt_xvbrp
        WHERE vbeln = gt_xvbrp-vgbel.
    SORT gt_likp BY vbeln.
  ENDIF.


  ASSIGN gt_likp TO FIELD-SYMBOL(<lt_likp>).

  LOOP AT gt_xvbrk INTO gs_xvbrk.

    LOOP AT gt_xvbrp ASSIGNING <vbrp>.
      CLEAR:  gs_lineprice.
*---- Pricing components

      gs_lineprice-vbeln = <vbrp>-vbeln.
      gs_lineprice-posnr = <vbrp>-posnr.
      gs_lineprice-fkimg = <vbrp>-fkimg.
      IF NOT <vbrp>-j_3ausize IS INITIAL.

      ENDIF.   "not <VBRP>-j_3ausize is initial

      PERFORM determine_prices_item  USING gs_xvbrk-knumv

                                            gs_lineprice-other

      APPEND gs_lineprice TO gt_lineprice.
    ENDLOOP.   "AT GT_VBRP ASSIGNING <VBRP>

      ENDIF.
      CLEAR gs_vbep.
      READ TABLE gt_vbep INTO gs_vbep WITH KEY vbeln = gs_xvbrp-aubel
                                               posnr = gs_xvbrp-aupos
                                               etenr = gs_xvbrp-j_3avgete BINARY SEARCH.
      PERFORM load_item.

    ENDLOOP.

    CLEAR gs_likp.
    LOOP AT gt_likp INTO gs_likp.           "loop at likp becasue there can be more than 1 delivery per invoice.
      PERFORM load_header .
    ENDLOOP.
    IF sy-subrc <> 0.
      PERFORM load_header .
    ENDIF.

  ENDLOOP.

  MOVE  gt_zsxx_pr_invoice[] TO inv_hdr[].
  MOVE  gt_zsxx_pr_invline[] TO inv_lines[].
