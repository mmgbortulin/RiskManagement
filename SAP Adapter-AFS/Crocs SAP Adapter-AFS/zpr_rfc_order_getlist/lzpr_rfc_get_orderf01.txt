*&      Form  GET_SALES_ORDERS
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_GT_VBMTV  text
*      -->P_SOLD_TO  text
*      -->P_FROM_DATE  text
*      -->P_TO_DATE  text
*      -->P_FROM_ORDER  text
*      -->P_TO_ORDER  text
*----------------------------------------------------------------------*
*form get_sales_orders  tables   pt_vbmtv structure svbmtv_trvog
FORM get_sales_orders  TABLES   pr_sales_org STRUCTURE zspr_vkorg
                                pr_dis_ch    STRUCTURE zspr_vtweg
                                pr_division  STRUCTURE zspr_spart
                                pr_ord_type  STRUCTURE zspr_auart
                       USING    p_sold_to
                                p_from_date
                                p_to_date
                                p_from_order
                                p_to_order
                                p_open_orders
                                p_idoc_search
                                p_from_weborder
                                p_to_weborder
                                p_custpo
                                p_style
                                p_ship_to
                                p_from_ship_date
                                p_to_ship_date.

  DATA: l_open_orders TYPE char1.

  l_open_orders = p_open_orders.

*SOLD TO
  IF p_sold_to IS NOT INITIAL.
    CLEAR: gr_kunnr[],
           gs_kunnr.
    gs_kunnr-sign = 'I'.
    gs_kunnr-option = 'EQ'.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = p_sold_to
      IMPORTING
        output = gs_kunnr-low.
    APPEND gs_kunnr TO gr_kunnr.
  ENDIF.

*DATE
  CLEAR gs_erdat.
  READ TABLE gr_erdat INTO gs_erdat INDEX 1.
  gs_erdat-sign = 'I'.
  gs_erdat-option = 'BT'.
  IF p_from_date IS NOT INITIAL.
    gs_erdat-low = p_from_date.
  ENDIF.
  IF p_to_date IS NOT INITIAL.
    gs_erdat-high = p_to_date.
  ELSE.
    gs_erdat-high = sy-datum.
  ENDIF.
  REFRESH gr_erdat.
  APPEND gs_erdat TO gr_erdat.

  IF p_idoc_search IS NOT INITIAL.
    PERFORM get_idoc_order TABLES gr_vbeln
                          USING p_from_order
                                p_to_order.
  ELSE.
*order No
    IF p_from_order IS NOT INITIAL OR p_to_order IS NOT INITIAL.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = p_from_order
        IMPORTING
          output = p_from_order.

      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = p_to_order
        IMPORTING
          output = p_to_order.
      gs_vbeln-sign = 'I'.
      gs_vbeln-option = 'EQ'.
      gs_vbeln-low = p_from_order.
      gs_vbeln-high = p_to_order.
      APPEND gs_vbeln TO gr_vbeln.
    ENDIF.
  ENDIF.

*Sales Org
  IF pr_sales_org[] IS NOT INITIAL.
    CLEAR: gr_vkorg[],
           gs_vkorg.
    LOOP AT pr_sales_org.
      gs_vkorg-sign = 'I'.
      gs_vkorg-option = 'EQ'.
      gs_vkorg-low = pr_sales_org-vkorg.
      APPEND gs_vkorg TO gr_vkorg.
    ENDLOOP.
  ENDIF.

*Dist Channel
  IF pr_dis_ch[] IS NOT INITIAL.
    CLEAR: gr_vtweg[],
           gs_vtweg.
    LOOP AT pr_dis_ch.
      gs_vtweg-sign = 'I'.
      gs_vtweg-option = 'EQ'.
      gs_vtweg-low = pr_dis_ch-vtweg.
      APPEND gs_vtweg TO gr_vtweg.
    ENDLOOP.
  ENDIF.

*Division
  IF pr_division[] IS NOT INITIAL.
    CLEAR: gr_spart[],
           gs_spart.
    LOOP AT pr_division.
      gs_spart-sign = 'I'.
      gs_spart-option = 'EQ'.
      gs_spart-low = pr_division-spart.
      APPEND gs_spart TO gr_spart.
    ENDLOOP.
  ENDIF.

*Order Type
  IF pr_ord_type[] IS NOT INITIAL.
    CLEAR: gr_auart[],
           gs_auart.
    LOOP AT pr_ord_type.
      gs_auart-sign = 'I'.
      gs_auart-option = 'EQ'.
      gs_auart-low = pr_ord_type-auart.
      APPEND gs_auart TO gr_auart.
    ENDLOOP.
  ENDIF.

*Material
  IF p_style IS NOT INITIAL.
    IF p_style CA '*'.
      gs_matnr-sign   = 'I'.
      gs_matnr-option = 'CP'.
    ELSE.
      gs_matnr-sign   = 'I'.
      gs_matnr-option = 'EQ'.
    ENDIF.
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
      EXPORTING
        input  = p_style
      IMPORTING
        output = gs_matnr-low.
    APPEND gs_matnr TO gr_matnr.
  ENDIF.

*SHIP DATE
  IF p_from_ship_date IS NOT INITIAL OR p_to_ship_date IS NOT INITIAL.
    CLEAR: gr_vdatu[],
           gs_vdatu.
    gs_vdatu-option = 'BT'.
    gs_vdatu-sign = 'I'.
    gs_vdatu-low = p_from_ship_date.
    IF p_to_ship_date IS INITIAL.
      gs_vdatu-high = '99999999'.
    ELSE.
      gs_vdatu-high = p_to_ship_date.
    ENDIF.
    APPEND gs_vdatu TO gr_vdatu.
  ENDIF.

*Ship TO
  IF p_ship_to IS NOT INITIAL.
    CLEAR: gr_shipto[],
           gs_shipto.
    gs_shipto-sign = 'I'.
    gs_shipto-option = 'EQ'.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = p_ship_to
      IMPORTING
        output = gs_shipto-low.
    APPEND gs_shipto TO gr_shipto.
  ENDIF.

*CUST PO
  IF p_custpo IS NOT INITIAL.
    IF p_custpo CA '*'.
      gs_bstkd-option = 'CP'.
      gs_bstkd-sign   = 'I'.
    ELSE.
      gs_bstkd-option = 'EQ'.
      gs_bstkd-sign   = 'I'.
    ENDIF.
    gs_bstkd-low = p_custpo.
    APPEND gs_bstkd TO gr_bstkd.
  ENDIF.

*web order No
  IF p_from_weborder IS NOT INITIAL OR p_to_weborder IS NOT INITIAL.
    IF p_from_weborder CA '*'.
      gs_ihrez-sign   = 'I'.
      gs_ihrez-option = 'CP'.
    ELSE.
      gs_ihrez-sign   = 'I'.
      gs_ihrez-option = 'EQ'.
    ENDIF.
    gs_ihrez-low = p_from_weborder.
    gs_ihrez-high = p_to_weborder.
    APPEND gs_ihrez TO gr_ihrez.
  ELSE.
    gr_ihrez[] = gr_ihrez[].
  ENDIF.

  PERFORM get_open_orders TABLES gr_vbeln
                                 gt_open_qty
                                 gt_order
                          USING  l_open_orders.

ENDFORM.                    " GET_SALES_ORDERS

FORM load_ord_header  TABLES   pt_order STRUCTURE gs_order
                               pt_open_qty STRUCTURE gs_open_qty
*                               pt_lvbmtv structure vbmtv
                               pt_ord_hdr
                      USING    p_website.

  TYPES: BEGIN OF typ_vbuk,
           vbeln TYPE vbeln,
           lfgsk TYPE vbuk-lfgsk,
           gbstk TYPE vbuk-gbstk,
         END OF typ_vbuk.

  DATA: lt_vbuk TYPE TABLE OF typ_vbuk,
        ls_vbuk TYPE typ_vbuk.

  DATA: ls_ord_hdr TYPE zsxx_pr_orderheader.

  DATA: ls_open_qty TYPE typ_open_qty,
        ls_order    TYPE typ_order.

  SELECT vbeln
         lfgsk
         gbstk FROM  vbuk INTO TABLE lt_vbuk
         FOR ALL ENTRIES IN pt_order
         WHERE  vbeln  = pt_order-vbeln.
  SORT lt_vbuk BY vbeln.

  SORT pt_order BY vbeln.

  LOOP AT pt_order INTO ls_order.

    CLEAR ls_ord_hdr.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        input  = ls_order-vbeln
      IMPORTING
        output = ls_ord_hdr-orderno.


    ls_ord_hdr-localeid = '1033'.
    ls_ord_hdr-hostref = ls_ord_hdr-orderno.
*    ls_ord_hdr-orderno              = LS_ORDER-vbeln.
    ls_ord_hdr-website              = p_website.
    ls_ord_hdr-salesorg             = ls_order-vkorg.
    ls_ord_hdr-division             = ls_order-spart.
    ls_ord_hdr-ordertype            = ls_order-auart.
    ls_ord_hdr-custpotype           = ls_order-bsark.
*    ls_ord_hdr-custpono             = ls_order-bstnk.  "20 char
    READ TABLE gt_vbkd ASSIGNING FIELD-SYMBOL(<vbkd>) WITH KEY vbeln = ls_order-vbeln BINARY SEARCH.
    IF sy-subrc = 0.
      ls_ord_hdr-custpono = <vbkd>-bstkd.
    ENDIF.
*    ls_ord_hdr-custpono             = ls_order-bstdk.   "35 char
    ls_ord_hdr-webordertype         = ls_order-bsark.
    ls_ord_hdr-weborderno           = ls_order-ihrez.
*    perform get_tracking_number tables lt_vbfa_del
*                                       lt_vekp
*                                changing ls_ord_hdr-trackingno.
*      ls_ord_hdr-trackingno           = LS_ORDER-inhalt.
    PERFORM format_timestamp  USING ls_order-audat
                                    gc_timefinish
                           CHANGING ls_ord_hdr-dateordered.
    PERFORM format_timestamp  USING ls_order-vdatu
                                    gc_timefinish
                           CHANGING ls_ord_hdr-daterequested.
*ls_ord_hdr-DATESHIPNOTBEFOR =
*ls_ord_hdr-DATESHIPNOTAFTER =
    PERFORM format_timestamp  USING ls_order-gueen
                                    gc_timefinish
                           CHANGING ls_ord_hdr-datecancelafter.
*      perform get_customer_details using LS_ORDER-kunnr
*                                         LS_ORDER-adrnr
*                                   changing gs_kna1.
*Sold To
*    ls_ord_hdr-soldto               = LS_ORDER-kunnr.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        input  = ls_order-kunnr
      IMPORTING
        output = ls_ord_hdr-soldto.

*    clear ls_partner.
*    read table lt_partner into ls_partner with key kunnr =  ls_order-kunnr.
*    ls_ord_hdr-soldtoname           = ls_partner-name1.
*    ls_ord_hdr-soldtoaddress1       = gs_kna1-address1.
*    ls_ord_hdr-soldtoaddress2       = gs_kna1-address2.
*    ls_ord_hdr-soldtoaddress3       = gs_kna1-address3.
*    ls_ord_hdr-soldtocity           = gs_kna1-city.
*    ls_ord_hdr-soldtoregion         = gs_kna1-region.
*    ls_ord_hdr-soldtopostalcode     = gs_kna1-postalcode.
*    ls_ord_hdr-soldtocountry        = gs_kna1-country.
    ls_ord_hdr-buyer                = ls_order-bname.
*ls_ord_hdr-EMAIL =
*Ship To
*    clear ls_partner.
*    read table lt_partner into ls_partner with key vbeln = ls_order-vbeln posnr = '000000' parvw = 'WE' binary search.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        input  = ls_order-shipto
      IMPORTING
        output = ls_ord_hdr-shipto.
*    ls_ord_hdr-storeno =
*    ls_ord_hdr-shiptoname           = LS_ORDER-zzname1_twe.
*    ls_ord_hdr-shiptoaddress1       = gs_kna1-address1.
*    ls_ord_hdr-shiptoaddress2       = gs_kna1-address2.
*    ls_ord_hdr-shiptoaddress3       = gs_kna1-address3.
*    ls_ord_hdr-shiptocity           = gs_kna1-city.
*    ls_ord_hdr-shiptoregion         = gs_kna1-region.
*    ls_ord_hdr-shiptopostalcode     = gs_kna1-postalcode.
*    ls_ord_hdr-shiptocountry        = gs_kna1-country.
*    clear ls_partner.
*    read table lt_partner into ls_partner with key vbeln =  ls_order-vbeln posnr = '000000' parvw = 'ZS' binary search.
*    ls_ord_hdr-storeno = ls_partner-kunnr..
*Bill To
*    clear ls_vbpa.
*    read table lt_vbpa into ls_vbpa with key vbeln =  LS_ORDER-vbeln posnr = '000000' parvw = 'RE' binary search.
*    perform get_customer_details using    ls_vbpa-kunnr
*                                          ls_vbpa-adrnr
*                                 changing gs_kna1.
*    ls_ord_hdr-billto               = gs_kna1-kunnr.
*    ls_ord_hdr-billtoname           = gs_kna1-name.
*    ls_ord_hdr-billtoaddress1       = gs_kna1-address1.
*    ls_ord_hdr-billtoaddress2       = gs_kna1-address1.
*    ls_ord_hdr-billtoaddress3       = gs_kna1-address3.
*    ls_ord_hdr-billtocity           = gs_kna1-city.
*    ls_ord_hdr-billtoregion         = gs_kna1-region.
*    ls_ord_hdr-billtopostalcode     = gs_kna1-postalcode.
*    ls_ord_hdr-billtocountry        = gs_kna1-country.
*    ls_ord_hdr-currency             = ls_order-waerk.
    ls_ord_hdr-creditstatus          = ls_order-cmgst.

    CLEAR ls_vbuk.
    READ TABLE lt_vbuk INTO ls_vbuk WITH KEY vbeln = ls_order-vbeln BINARY SEARCH.
    IF ls_vbuk-lfgsk = 'C' AND ls_vbuk-gbstk = 'C'.
      ls_ord_hdr-orderstatus         = 'COMPLETE'.
    ELSEIF ls_vbuk-lfgsk = 'B' AND ls_vbuk-gbstk = 'B'.
      ls_ord_hdr-orderstatus         = 'PARTIAL'.
    ELSE.
      ls_ord_hdr-orderstatus         = 'OPEN'.
    ENDIF.
*    ls_ord_hdr-termscode            = ls_vbkd-zterm.
*    ls_ord_hdr-termsdescription     = ls_vbkd-text1.
*    clear ls_partner.
*    read table lt_partner into ls_partner with key vbeln =  ls_order-vbeln posnr = '000000' parvw = 'ZD' binary search.
*    if sy-subrc <> 0.
*      read table lt_partner into ls_partner with key vbeln =  ls_order-vbeln posnr = '000000' parvw = 'XD' binary search.
*    endif.
*    ls_ord_hdr-shipmethodcode = ls_partner-kunnr.
*    ls_ord_hdr-shipmethodtext = ls_partner-name1.
*    clear ls_kna1.
*    read table lt_kna1 into ls_kna1 with key kunnr = ls_vbpa-kunnr binary search.
*    ls_ord_hdr-freighttermscode = ls_vbkd-inco1.
*    ls_ord_hdr-freighttermstext = ls_vbkd-bezei.
*      clear ls_tinct.
*      read table lt_tinct into ls_tinct with key inco1 = ls_vbkd-inco1 binary search.
*    clear ls_ord_hdr-totalunits.
*    read table lt_tot_units into ls_tot_units with key vbeln = LS_ORDER-vbeln binary search.
    CLEAR ls_open_qty.
    READ TABLE pt_open_qty INTO ls_open_qty WITH KEY vbeln = ls_order-vbeln BINARY SEARCH.
    ls_ord_hdr-totalunits           = ls_open_qty-qty.
    IF ls_ord_hdr-totalunits < 0.
      ls_ord_hdr-totalunits = 0.
    ENDIF.
*    ls_ord_hdr-discountwholesal =
*    ls_ord_hdr-subtotalwholesal =
*-- AFS Material
    IF ls_open_qty-j_3anetw IS NOT INITIAL.
      WRITE: ls_open_qty-j_3anetw TO ls_ord_hdr-subtotalnet CURRENCY ls_order-waerk NO-GROUPING.
*-- STD Material
    ELSE.
      WRITE: ls_order-netwr TO ls_ord_hdr-subtotalnet
                       CURRENCY ls_order-waerk
                       NO-GROUPING.
    ENDIF.
*    endloop.

*      ls_ord_hdr-subtotalnet          = LS_ORDER-netwr.
*    ls_ord_hdr-dateinserted =
    CONCATENATE sy-datum sy-uzeit INTO ls_ord_hdr-dateinserted.
    APPEND ls_ord_hdr TO pt_ord_hdr.
  ENDLOOP.



ENDFORM.                    " LOAD_ORD_HEADER
*&---------------------------------------------------------------------*
*&      Form  FORMAT_TIMESTAMP
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_LS_VBAK_GUEEN  text
*      -->P_GC_TIMEFINISH  text
*      <--P_LS_ORD_HDR_DATECANCELAFTER  text
*----------------------------------------------------------------------*
FORM format_timestamp USING    us_date TYPE sydatum
                               us_time TYPE syuzeit
                      CHANGING ch_timestamp.
* FORMAT TIMESTAMP FROM DATE (CCYYMMDD) AND TIME (HHMMSS).

  IF us_date IS INITIAL.
*-- NO DATE... SO RETURN BLANK TIMESTAMP VALUE
    CLEAR: ch_timestamp.
  ELSE.
*-- FORMAT DATE/TIME
    CONCATENATE us_date
                us_time
           INTO ch_timestamp.
  ENDIF.

ENDFORM.                    " FORMAT_TIMESTAMP

*&      Form  GET_CUSTOMER_DETAILS
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_LS_VBTMV_ADRNR  text
*      <--P_LS_KNA1  text
*----------------------------------------------------------------------*
FORM get_customer_details USING    p_kunnr
                                   p_adrnr
                          CHANGING ch_kna1 LIKE gs_kna1.
* DESCRIPTION:  Get customer address
*// Local Declaration
  DATA:  lv_tabix(1)    TYPE n,                "table index
         lx_street,                         "X: street found
         lv_returnlines TYPE adrs-anzzl,
         lt_adrtable    TYPE  szadr_printform_table.

  DATA: BEGIN OF ls_adrc,
          addrnumber TYPE adrc-addrnumber,
          name1      TYPE adrc-name1,
          name2      TYPE adrc-name2,
          city1      TYPE adrc-city1,
          post_code1 TYPE adrc-post_code1,
          street     TYPE adrc-street,
          str_suppl1 TYPE adrc-str_suppl1,
          str_suppl2 TYPE adrc-str_suppl2,
          str_suppl3 TYPE adrc-str_suppl3,
          location   TYPE adrc-location,
          country    TYPE adrc-country,
          langu      TYPE adrc-langu,
          region     TYPE adrc-region,
        END OF ls_adrc.


  FIELD-SYMBOLS:  <adr> LIKE LINE OF lt_adrtable.

*-- Read for this customer/address
  CLEAR: lx_street,
         lv_tabix,
         lv_returnlines,
         lt_adrtable[].
  CALL FUNCTION 'ADDRESS_INTO_PRINTFORM'
    EXPORTING
      address_type            = '1'
      address_number          = p_adrnr
      number_of_lines         = 6
*     STREET_HAS_PRIORITY     = ' '
    IMPORTING
      number_of_used_lines    = lv_returnlines
*     ADDRESS_NOT_FOUND       =
      address_printform_table = lt_adrtable.
  DELETE lt_adrtable WHERE line_type = 'O'  OR    "city/state/zip
                           line_type = 'L'.       "country

  LOOP AT lt_adrtable ASSIGNING <adr>.
    lv_tabix = sy-tabix.
    IF lv_tabix > 4.
      EXIT.
    ELSE.
      IF <adr>-line_type = 'S'.
        lx_street = 'X'.
      ENDIF.
*------ map name/address fields
      CASE lv_tabix.
        WHEN '1'.
          ch_kna1-name     = <adr>-address_line.
        WHEN '2'.
          ch_kna1-address1 = <adr>-address_line.
        WHEN '3'.
          ch_kna1-address2 = <adr>-address_line.
        WHEN '4'.
          ch_kna1-address3 = <adr>-address_line.
        WHEN OTHERS.
      ENDCASE.
    ENDIF.
  ENDLOOP.   "AT LT_ADRTABLE assigning <ADR>
  IF lx_street IS INITIAL.
*---- street missing... need to find street.
    READ TABLE lt_adrtable ASSIGNING <adr>
                           WITH KEY line_type = 'S'.
    IF sy-subrc = 0.
      ch_kna1-address3 = <adr>-address_line.
    ENDIF.   "sy-subrc = 0 [on read table lt_adrtable...]
  ENDIF.   "lx_street is initial
*-- Get rest of address details from ADRC
  SELECT SINGLE  addrnumber name1 name2
                 city1 post_code1 street
                 str_suppl1 str_suppl2 str_suppl3
                 location country langu
                 region
            FROM adrc INTO ls_adrc
            WHERE addrnumber = p_adrnr.
  IF sy-subrc = 0.
    ch_kna1-kunnr      = p_kunnr.
    ch_kna1-adrnr      = p_adrnr.
    ch_kna1-city       = ls_adrc-city1.
    ch_kna1-region     = ls_adrc-region.
    ch_kna1-postalcode = ls_adrc-post_code1.
    ch_kna1-country    = ls_adrc-country.
  ENDIF.   "sy-subrc = 0 [ON SELECT SINGLE...FROM ADRC]

ENDFORM.                    " GET_CUSTOMER_DETAILS

*      -->P_GS_ORDER_SCHEDULES_OUT_BMENG  text
*      <--P_LV_WHOLESALE  text
*      <--P_LV_WHOLESALE_UNITPRICE  text
*      <--P_LV_DISCOUNT  text
*      <--P_LV_TAXES  text
*      <--P_LV_PERCENT  text
*      <--P_LV_OTHER  text
*----------------------------------------------------------------------*
FORM determine_prices_item  TABLES  pt_konv STRUCTURE konv
                            USING   us_posnr TYPE posnr
                                    us_etenr TYPE etenr
                                    us_kbmeng TYPE vbap-kbmeng
                                    us_bmeng TYPE vbep-bmeng
                           CHANGING ch_wholesale  "wholesale amount
                                    ch_unitprice  "unit price wholesale
                                    ch_discount   "discount amount
                                    ch_taxes      "taxes
                                    ch_percent    "discount percent
                                    ch_other.     "other

*  DATA: l_kbetr TYPE konv-kbetr.

  FIELD-SYMBOLS: <konv> LIKE konv.
* DESCRIPTION:  Assign pricing components

* Begin processing
  CLEAR:  ch_wholesale,
          ch_unitprice,
          ch_discount,
          ch_taxes,
          ch_other.

* check elements at line item level
  LOOP AT pt_konv ASSIGNING <konv>
                  WHERE kposn = us_posnr     AND
                        j_3aetenr = '0000'   AND
                        kstat = space        AND
                        kinak = space.
*-- check Condtion Class... prices, taxes, discounts, etc.
    CASE <konv>-koaid.
      WHEN 'B'.       "prices
        IF <konv>-kntyp = 'H' OR <konv>-kntyp = space.
*-------- wholesale amount
          IF us_kbmeng > 0.
            ch_wholesale = ch_wholesale + ( <konv>-kwert * us_bmeng / us_kbmeng ).
          ELSE.
            ch_wholesale = ch_wholesale + ( <konv>-kwert * us_bmeng ).
          ENDIF.
          IF <konv>-krech = 'C'.
*---------- wholesale unit price
            ch_unitprice = <konv>-kbetr.
          ENDIF.
        ELSE.
*-------- OTHER Amount
          IF us_kbmeng > 0.
            ch_other = ch_other + ( <konv>-kwert * us_bmeng / us_kbmeng ).
          ELSE.
            ch_other = ch_other + ( <konv>-kwert * us_bmeng ).
          ENDIF.
        ENDIF.   "<KONV>-KSTAT = space...
      WHEN 'A'.       "Discounts/Surcharges
        IF <konv>-kntyp = space.
*-------- (-): discount; (+): surcharge
          IF us_kbmeng > 0.
            ch_discount = ch_discount + ( <konv>-kwert * us_bmeng / us_kbmeng ).
          ELSE.
            ch_discount = ch_discount + ( <konv>-kwert * us_bmeng ).
          ENDIF.
          IF <konv>-krech CA 'AHI'.
            IF ch_percent IS INITIAL.
              WRITE:  <konv>-kbetr TO ch_percent
                                   CURRENCY '3'
                                   NO-GROUPING.
            ELSE.
              ch_percent = 'CALC'.
            ENDIF.   "CH_PERCENT is initial
          ELSE.
            ch_percent = 'CALC'.
          ENDIF.   "<KONV>-KRECH ca 'AHI'
        ELSE.
*-------- OTHER Amount
          ch_other = ch_other + ( <konv>-kwert * us_bmeng / us_kbmeng ).
        ENDIF.   "<KONV>-KNTYP = space
      WHEN 'D'.       "TAXES
*------ Taxes
        ch_taxes = ch_taxes + ( <konv>-kwert * us_bmeng / us_kbmeng ).
*      WHEN 'C'.       "Expense reimbursement/rebate
**------ Rebates??
*        CH_Rebate = CH_Rebate + <KONV>-KWERT.
      WHEN OTHERS.
*------ OTHER Amount
        ch_other = ch_other + ( <konv>-kwert * us_bmeng / us_kbmeng ).
    ENDCASE.   "<KONV>-KOAID
  ENDLOOP.   "AT GT_KONV assigning <KONV>

* 2nd pass to check elements at schedule line level
  LOOP AT pt_konv ASSIGNING <konv>
                  WHERE kposn = us_posnr        AND
                        j_3aetenr = us_etenr    AND
                        kstat = space           AND
                        kinak = space.
*-- check Condtion Class... prices, taxes, discounts, etc.
    CASE <konv>-koaid.
      WHEN 'B'.       "prices
        IF <konv>-kntyp = 'H'   OR
           <konv>-kntyp = space.
*-------- wholesale amount
          ch_wholesale = ch_wholesale +  <konv>-kwert.
          IF <konv>-krech = 'C'.
*---------- wholesale unit price
            ch_unitprice = <konv>-kbetr.
          ENDIF.
        ELSE.

  DATA:  l_open           TYPE menge_d,

                                            posnr = gs_order_schedules_out-itm_number

        l_open = ls_vbep-wmeng.
      ELSE.
        l_open = ls_vbep-bmeng.      "<-- initial assignment for open qty

                     WHERE vbelv      = p_orderno  AND
                           posnv      = gs_order_schedules_out-itm_number  AND
                           j_3aetenr  = gs_order_schedules_out-sched_line  AND

    IF l_open > l_invoiced.
      l_order_open = 'X'.
    ENDIF.   "l_open > l_invoiced
    IF gs_order_itemstatus_out-dlv_stat_i = 'B'  AND l_open IS NOT INITIAL.
      PERFORM get_allocated_qty  USING p_orderno
                                       gs_order_schedules_out-itm_number
                                       gs_order_schedules_out-sched_line
                              CHANGING l_qty_base_unit.
      PERFORM convert_uom USING l_qty_base_unit
                                ls_vbep-umvkn
                                ls_vbep-umvkz
                      CHANGING  l_allocated.
      l_open = l_open - l_allocated. "<-- open, not allocated
    ENDIF.
    IF l_invoiced = ps_order_line-cum_cf_qty.          "ALL INVOICED
* Checking for line item is Rejected or not
      IF ls_vbep-j_3aabgru EQ space.

      ELSE.

      ps_ord_hdr_lines-qtyunavailable = ls_vbep-wmeng.

      l_allocated = 0.
    ELSE.
      l_allocated = ls_vbep-wmeng.
      ps_ord_hdr_lines-qtyunavailable = 0.
      IF gs_order_schdstatus_out-lfgsa = 'A'.
        p_status = 'OPEN'.
      ELSEIF gs_order_schdstatus_out-lfgsa = 'C' OR gs_order_schdstatus_out-lfgsa = 'B'.

                          WHERE vbelv      = p_orderno  AND

                                    ls_vbep-umvkz

          READ TABLE gt_delstatus INTO ls_delstatus WITH KEY vbeln = ls_vbfa-vbeln
                                                             posnr = ls_vbfa-posnn BINARY SEARCH.
          IF sy-subrc = 0.
            IF ls_delstatus-wbsta = 'C'.
*---------- shipped!
              l_shipped = l_shipped + l_qty_sales_unit.
              p_status = 'COMPLETE'.
            ELSEIF ls_delstatus-wbsta = 'A'.
              p_status = 'OPEN'.
            ENDIF.
          ENDIF.   "sy-subrc = 0 [on READ table GTS_DELSTATUS...]
        ENDLOOP.   "at gts_vbfa assigning <vbfa>
        IF p_status IS INITIAL.
          p_status = 'OPEN'.
        ENDIF.
      ENDIF.
      l_open = l_allocated - l_shipped.
    ENDIF.
  ENDIF.

* open qty
  WRITE:  l_open TO ps_ord_hdr_lines-qtyopen
                  UNIT ps_order_line-sales_unit NO-GROUPING.
** allocated qty
  WRITE:  l_allocated TO ps_ord_hdr_lines-qtyallocated
                  UNIT ps_order_line-sales_unit NO-GROUPING.
** Picked qty
  WRITE:  l_picked TO ps_ord_hdr_lines-qtypicked
                  UNIT ps_order_line-sales_unit NO-GROUPING.
** Shipped qty
  WRITE:  l_shipped TO ps_ord_hdr_lines-qtyshipped
                  UNIT ps_order_line-sales_unit NO-GROUPING.
* Qty Balance (Will appear on website as OPEN.  Include everything
*              except shipped.  Field shipped only includes invoiced).
  l_balance = l_allocated - l_shipped.
  WRITE:  l_balance TO ps_ord_hdr_lines-qtybalance
                  UNIT ps_order_line-sales_unit NO-GROUPING.

  p_invoiced = l_invoiced.
  p_open     = l_open.

                     USING    p_from_order_no
                              p_to_order_no.

  TYPES: BEGIN OF ltyp_idoc_order,
           docnum TYPE edi_docnum,

        ls_vbeln LIKE LINE OF lr_vbeln.

    EXPORTING

    EXPORTING
      input  = p_to_order_no
    IMPORTING
      output = l_to_idoc.


  SELECT edids~docnum edids~stapa2 INTO TABLE lt_idoc_order
         FROM  edids AS edids
         INNER JOIN edidc AS edidc
         ON edidc~docnum = edids~docnum
         WHERE  edids~docnum  BETWEEN l_from_idoc AND l_to_idoc

         AND    edidc~mestyp  = 'ORDERS'.

    MOVE ls_idoc_order-docnum+6(10) TO l_docnum.
    READ TABLE pr_vbeln TRANSPORTING NO FIELDS WITH KEY low = l_docnum BINARY SEARCH.
    IF sy-subrc = 0.
      DELETE pr_vbeln INDEX sy-tabix.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_so_qty INTO ls_so_qty.
    CLEAR ls_inv_qty.
*determine open or complete
    CASE p_open_orders.
      WHEN 'O'."Open Orders
        READ TABLE lt_inv_qty INTO ls_inv_qty WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
        IF sy-subrc = 0.
          IF ls_so_qty-bmeng > ls_inv_qty-rfmng.
            CLEAR ls_order.
            READ TABLE lt_order INTO ls_order WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
            APPEND ls_order TO pt_orders.
          ENDIF.
        ELSE.
          CLEAR ls_order.
          READ TABLE lt_order INTO ls_order WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
          APPEND ls_order TO pt_orders.
        ENDIF.
      WHEN 'C'."Complete Orders
        READ TABLE lt_inv_qty INTO ls_inv_qty WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
        IF sy-subrc = 0.
          IF ls_so_qty-bmeng <= ls_inv_qty-rfmng.
            CLEAR ls_order.
            READ TABLE lt_order INTO ls_order WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
            APPEND ls_order TO pt_orders.
          ENDIF.
        ENDIF.
      WHEN ' '.
        READ TABLE lt_inv_qty INTO ls_inv_qty WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
        CLEAR ls_order.
        READ TABLE lt_order INTO ls_order WITH KEY vbeln = ls_so_qty-vbeln BINARY SEARCH.
        APPEND ls_order TO pt_orders.
      WHEN OTHERS.
    ENDCASE.
*get open qty
    ls_open_qty-vbeln = ls_so_qty-vbeln.
    ls_open_qty-qty = ls_so_qty-bmeng - ls_inv_qty-rfmng.
    ls_open_qty-j_3anetw = ls_so_qty-j_3anetw.
    APPEND ls_open_qty TO pt_open_qty.
  ENDLOOP.

  SORT pt_orders BY vbeln.

  SORT pt_open_qty BY vbeln.

ENDFORM.                    " GET_OPEN_ORDERS
